class DeepSORT: def __init__(self, max_age=30, min_hits=3, iou_threshold=0.3): self.max_age = max_age self.min_hits = min_hits self.iou_threshold = iou_threshold self.tracks = [] self.track_id = 0 self.feature_extractor = ReIDNetwork() def update(self, detections, frame): # Extract appearance features features = self.feature_extractor(frame, detections) # Predict new locations of existing tracks for track in self.tracks: track.predict() # Compute cost matrix (IOU + appearance) cost_matrix = self.compute_cost_matrix( self.tracks, detections, features ) # Hungarian algorithm for optimal assignment matches, unmatched_dets, unmatched_tracks = \ linear_assignment(cost_matrix, self.iou_threshold) # Update matched tracks for track_idx, det_idx in matches: self.tracks[track_idx].update( detections[det_idx], features[det_idx] ) # Create new tracks for unmatched detections for det_idx in unmatched_dets: self.create_track(detections[det_idx], features[det_idx]) return self.get_active_tracks()
