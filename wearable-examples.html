Example 1: Complete Sensor Fusion Pipeline

Full implementation of Madgwick filter with magnetometer integration for 9-axis fusion.

import numpy as np
from scipy.spatial.transform import Rotation

class MadgwickFilter:
    """Madgwick AHRS filter for 9-axis IMU fusion."""

    def __init__(self, sample_rate=50, beta=0.1):
        self.sample_rate = sample_rate
        self.beta = beta  # Filter gain
        self.q = np.array([1.0, 0.0, 0.0, 0.0])  # Quaternion

    def update(self, accel, gyro, mag=None):
        """Update orientation with new sensor readings."""
        q = self.q
        dt = 1.0 / self.sample_rate

        # Normalize accelerometer
        a = accel / np.linalg.norm(accel)

        # Gradient descent step
        if mag is not None:
            # 9-axis fusion with magnetometer
            m = mag / np.linalg.norm(mag)
            h = self._quat_mult(q, self._quat_mult([0, m[0], m[1], m[2]],
                                                   self._quat_conj(q)))
            b = np.array([0, np.sqrt(h[1]**2 + h[2]**2), 0, h[3]])

            f = np.array([
                2*(q[1]*q[3] - q[0]*q[2]) - a[0],
                2*(q[0]*q[1] + q[2]*q[3]) - a[1],
                2*(0.5 - q[1]**2 - q[2]**2) - a[2],
                2*b[1]*(0.5 - q[2]**2 - q[3]**2) + 2*b[3]*(q[1]*q[3] - q[0]*q[2]) - m[0],
                2*b[1]*(q[1]*q[2] - q[0]*q[3]) + 2*b[3]*(q[0]*q[1] + q[2]*q[3]) - m[1],
                2*b[1]*(q[0]*q[2] + q[1]*q[3]) + 2*b[3]*(0.5 - q[1]**2 - q[2]**2) - m[2]
            ])
            J = self._jacobian_9dof(q, b)
        else:
            # 6-axis fusion
            f = np.array([
                2*(q[1]*q[3] - q[0]*q[2]) - a[0],
                2*(q[0]*q[1] + q[2]*q[3]) - a[1],
                2*(0.5 - q[1]**2 - q[2]**2) - a[2]
            ])
            J = self._jacobian_6dof(q)

        # Gradient step
        grad = J.T @ f
        grad = grad / np.linalg.norm(grad)

        # Quaternion derivative from gyroscope
        qDot = 0.5 * self._quat_mult(q, [0, gyro[0], gyro[1], gyro[2]])

        # Fuse and integrate
        qDot -= self.beta * grad
        self.q = q + qDot * dt
        self.q /= np.linalg.norm(self.q)

        return self.q

    def get_euler(self):
        """Get Euler angles (roll, pitch, yaw) in degrees."""
        q = self.q
        roll = np.arctan2(2*(q[0]*q[1] + q[2]*q[3]),
                          1 - 2*(q[1]**2 + q[2]**2))
        pitch = np.arcsin(2*(q[0]*q[2] - q[3]*q[1]))
        yaw = np.arctan2(2*(q[0]*q[3] + q[1]*q[2]),
                         1 - 2*(q[2]**2 + q[3]**2))
        return np.degrees([roll, pitch, yaw])

    def _quat_mult(self, q1, q2):
        return np.array([
            q1[0]*q2[0] - q1[1]*q2[1] - q1[2]*q2[2] - q1[3]*q2[3],
            q1[0]*q2[1] + q1[1]*q2[0] + q1[2]*q2[3] - q1[3]*q2[2],
            q1[0]*q2[2] - q1[1]*q2[3] + q1[2]*q2[0] + q1[3]*q2[1],
            q1[0]*q2[3] + q1[1]*q2[2] - q1[2]*q2[1] + q1[3]*q2[0]
        ])

    def _quat_conj(self, q):
        return np.array([q[0], -q[1], -q[2], -q[3]])

# Usage
fusion = MadgwickFilter(sample_rate=50, beta=0.1)
for accel, gyro, mag in sensor_stream:
    q = fusion.update(accel, gyro, mag)
    roll, pitch, yaw = fusion.get_euler()
    print(f"RPY: {roll:.1f}, {pitch:.1f}, {yaw:.1f}")

Example 2: Complete HRV Analysis

Time and frequency domain HRV analysis from PPG-derived RR intervals.

import numpy as np
from scipy import signal, interpolate

class HRVAnalyzer:
    """Complete HRV analysis from RR intervals."""

    def __init__(self, rr_intervals):
        """Initialize with RR intervals in milliseconds."""
        self.rr = np.array(rr_intervals)
        self.rr_clean = self._remove_ectopic()

    def _remove_ectopic(self, threshold=0.2):
        """Remove ectopic beats using threshold method."""
        rr = self.rr.copy()
        median_rr = np.median(rr)
        for i in range(1, len(rr) - 1):
            if abs(rr[i] - median_rr) / median_rr > threshold:
                rr[i] = (rr[i-1] + rr[i+1]) / 2
        return rr

    def time_domain(self):
        """Calculate time-domain HRV metrics."""
        rr = self.rr_clean
        diff_rr = np.diff(rr)

        return {
            'mean_rr': np.mean(rr),
            'sdnn': np.std(rr, ddof=1),
            'rmssd': np.sqrt(np.mean(diff_rr**2)),
            'pnn50': np.sum(np.abs(diff_rr) > 50) / len(diff_rr) * 100,
            'mean_hr': 60000 / np.mean(rr),
            'sdhr': np.std(60000 / rr)
        }

    def frequency_domain(self, fs=4):
        """Calculate frequency-domain HRV metrics using Welch method."""
        rr = self.rr_clean

        # Interpolate to uniform sampling
        t = np.cumsum(rr) / 1000  # Convert to seconds
        t -= t[0]
        f_interp = interpolate.interp1d(t, rr, kind='cubic')
        t_uniform = np.arange(t[0], t[-1], 1/fs)
        rr_uniform = f_interp(t_uniform)

        # Remove mean and compute PSD
        rr_uniform -= np.mean(rr_uniform)
        freqs, psd = signal.welch(rr_uniform, fs=fs, nperseg=len(rr_uniform)//2)

        # Frequency bands
        vlf_mask = (freqs >= 0.003) & (freqs < 0.04)
        lf_mask = (freqs >= 0.04) & (freqs < 0.15)
        hf_mask = (freqs >= 0.15) & (freqs < 0.4)

        vlf = np.trapz(psd[vlf_mask], freqs[vlf_mask])
        lf = np.trapz(psd[lf_mask], freqs[lf_mask])
        hf = np.trapz(psd[hf_mask], freqs[hf_mask])

        return {
            'vlf_power': vlf,
            'lf_power': lf,
            'hf_power': hf,
            'lf_hf_ratio': lf / hf if hf > 0 else np.nan,
            'total_power': vlf + lf + hf,
            'lf_nu': lf / (lf + hf) * 100,
            'hf_nu': hf / (lf + hf) * 100
        }

    def poincare(self):
        """Calculate Poincare plot metrics."""
        rr = self.rr_clean
        rr_n = rr[:-1]
        rr_n1 = rr[1:]

        sd1 = np.std(rr_n1 - rr_n) / np.sqrt(2)
        sd2 = np.std(rr_n1 + rr_n) / np.sqrt(2)

        return {
            'sd1': sd1,
            'sd2': sd2,
            'sd_ratio': sd1 / sd2 if sd2 > 0 else np.nan,
            'ellipse_area': np.pi * sd1 * sd2
        }

# Usage
rr_intervals = [820, 835, 810, 845, 825, 815, ...]
analyzer = HRVAnalyzer(rr_intervals)

time_metrics = analyzer.time_domain()
freq_metrics = analyzer.frequency_domain()
poincare_metrics = analyzer.poincare()

print(f"RMSSD: {time_metrics['rmssd']:.1f} ms")
print(f"LF/HF: {freq_metrics['lf_hf_ratio']:.2f}")

Example 3: Real-Time Activity Recognition

Streaming activity classification with TensorFlow Lite inference.

import numpy as np
import tensorflow as tf
from collections import deque

class StreamingActivityClassifier:
    """Real-time activity classification with sliding window."""

    ACTIVITIES = [
        'Walking', 'Running', 'Sitting', 'Standing',
        'Stairs_Up', 'Stairs_Down', 'Cycling', 'Sleeping'
    ]

    def __init__(self, model_path, window_size=128, overlap=0.5):
        self.window_size = window_size
        self.step_size = int(window_size * (1 - overlap))

        # Load TFLite model
        self.interpreter = tf.lite.Interpreter(model_path=model_path)
        self.interpreter.allocate_tensors()
        self.input_details = self.interpreter.get_input_details()
        self.output_details = self.interpreter.get_output_details()

        # Sliding window buffer
        self.buffer = deque(maxlen=window_size)
        self.sample_count = 0

        # Smoothing with moving average
        self.prediction_history = deque(maxlen=5)

    def preprocess(self, accel, gyro):
        """Normalize sensor data."""
        # Combine accelerometer and gyroscope
        sample = np.concatenate([accel, gyro])

        # Normalize (assuming ±4g accel, ±500°/s gyro)
        sample[:3] /= 39.2  # 4g in m/s²
        sample[3:] /= 8.73  # 500°/s in rad/s

        return sample

    def add_sample(self, accel, gyro):
        """Add new sensor sample and classify if window ready."""
        sample = self.preprocess(accel, gyro)
        self.buffer.append(sample)
        self.sample_count += 1

        # Check if we have enough samples
        if len(self.buffer) < self.window_size:
            return None

        # Classify every step_size samples
        if self.sample_count >= self.step_size:
            self.sample_count = 0
            return self._classify()

        return None

    def _classify(self):
        """Run inference on current window."""
        window = np.array(self.buffer, dtype=np.float32)
        window = np.expand_dims(window, axis=0)  # Add batch dim

        # Quantize if model is int8
        input_scale = self.input_details[0]['quantization'][0]
        input_zero = self.input_details[0]['quantization'][1]

        if input_scale != 0:
            window = (window / input_scale + input_zero).astype(np.int8)

        # Run inference
        self.interpreter.set_tensor(self.input_details[0]['index'], window)
        self.interpreter.invoke()
        output = self.interpreter.get_tensor(self.output_details[0]['index'])

        # Dequantize output
        output_scale = self.output_details[0]['quantization'][0]
        output_zero = self.output_details[0]['quantization'][1]

        if output_scale != 0:
            output = (output.astype(np.float32) - output_zero) * output_scale

        # Smooth predictions
        self.prediction_history.append(output[0])
        avg_probs = np.mean(self.prediction_history, axis=0)

        activity_id = np.argmax(avg_probs)
        confidence = avg_probs[activity_id]

        return {
            'activity': self.ACTIVITIES[activity_id],
            'activity_id': activity_id,
            'confidence': confidence,
            'probabilities': dict(zip(self.ACTIVITIES, avg_probs))
        }

# Usage
classifier = StreamingActivityClassifier('activity_int8.tflite')

while True:
    accel, gyro = imu.read()
    result = classifier.add_sample(accel, gyro)

    if result:
        print(f"{result['activity']}: {result['confidence']:.1%}")

Example 4: Overnight Sleep Analysis

Complete sleep staging and quality assessment from overnight sensor data.

import numpy as np
from sklearn.ensemble import RandomForestClassifier
from datetime import datetime, timedelta

class SleepAnalyzer:
    """Sleep staging and quality analysis."""

    STAGES = ['Awake', 'REM', 'Light', 'Deep']

    def __init__(self, model_path=None, epoch_duration=30):
        self.epoch_duration = epoch_duration  # seconds
        self.model = self._load_model(model_path)

    def _load_model(self, path):
        if path:
            import joblib
            return joblib.load(path)
        # Default simple model
        return RandomForestClassifier(n_estimators=100)

    def extract_epoch_features(self, accel_epoch, hr_epoch):
        """Extract features from single epoch."""
        # Activity features
        accel_mag = np.linalg.norm(accel_epoch, axis=1)
        activity_count = np.sum(np.abs(np.diff(accel_mag)))
        movement_var = np.var(accel_epoch)

        # HR features
        hr_mean = np.mean(hr_epoch)
        hr_std = np.std(hr_epoch)
        hr_range = np.max(hr_epoch) - np.min(hr_epoch)

        return [
            activity_count, movement_var,
            hr_mean, hr_std, hr_range
        ]

    def analyze(self, accel_data, hr_data, sample_rate_accel=50,
               sample_rate_hr=1, start_time=None):
        """Analyze overnight data and return sleep metrics."""

        samples_per_epoch_accel = self.epoch_duration * sample_rate_accel
        samples_per_epoch_hr = self.epoch_duration * sample_rate_hr

        n_epochs = min(
            len(accel_data) // samples_per_epoch_accel,
            len(hr_data) // samples_per_epoch_hr
        )

        stages = []
        features_list = []

        for i in range(n_epochs):
            accel_epoch = accel_data[
                i*samples_per_epoch_accel:(i+1)*samples_per_epoch_accel
            ]
            hr_epoch = hr_data[
                i*samples_per_epoch_hr:(i+1)*samples_per_epoch_hr
            ]

            features = self.extract_epoch_features(accel_epoch, hr_epoch)
            features_list.append(features)

        # Predict stages
        features_array = np.array(features_list)
        stages = self.model.predict(features_array)

        # Compute metrics
        return self._compute_metrics(stages, start_time)

    def _compute_metrics(self, stages, start_time):
        """Compute sleep quality metrics."""
        n_epochs = len(stages)
        epoch_min = self.epoch_duration / 60

        # Time in each stage
        awake_epochs = np.sum(stages == 0)
        rem_epochs = np.sum(stages == 1)
        light_epochs = np.sum(stages == 2)
        deep_epochs = np.sum(stages == 3)

        sleep_epochs = rem_epochs + light_epochs + deep_epochs
        total_sleep_min = sleep_epochs * epoch_min

        # Sleep efficiency
        efficiency = sleep_epochs / n_epochs if n_epochs > 0 else 0

        # Sleep latency (time to first sleep)
        sleep_onset = np.argmax(stages > 0) * epoch_min

        # WASO (Wake After Sleep Onset)
        if np.any(stages > 0):
            first_sleep = np.argmax(stages > 0)
            last_sleep = len(stages) - 1 - np.argmax(stages[::-1] > 0)
            waso = np.sum(stages[first_sleep:last_sleep+1] == 0) * epoch_min
        else:
            waso = 0

        # Sleep score (0-100)
        score = self._calculate_score(
            total_sleep_min, efficiency, deep_epochs * epoch_min,
            rem_epochs * epoch_min, waso
        )

        return {
            'total_sleep_min': total_sleep_min,
            'efficiency': efficiency,
            'sleep_latency_min': sleep_onset,
            'waso_min': waso,
            'deep_min': deep_epochs * epoch_min,
            'light_min': light_epochs * epoch_min,
            'rem_min': rem_epochs * epoch_min,
            'awake_min': awake_epochs * epoch_min,
            'score': score,
            'stages': stages.tolist()
        }

    def _calculate_score(self, total, efficiency, deep, rem, waso):
        """Calculate sleep score 0-100."""
        score = 0

        # Duration (40 points max)
        if total >= 420:  # 7 hours
            score += 40
        else:
            score += (total / 420) * 40

        # Efficiency (30 points max)
        score += efficiency * 30

        # Deep sleep (15 points max)
        deep_target = total * 0.20  # 20% target
        score += min(deep / deep_target, 1) * 15 if deep_target > 0 else 0

        # REM (15 points max)
        rem_target = total * 0.20  # 20% target
        score += min(rem / rem_target, 1) * 15 if rem_target > 0 else 0

        return min(100, int(score))

# Usage
analyzer = SleepAnalyzer()
result = analyzer.analyze(overnight_accel, overnight_hr)
print(f"Sleep Score: {result['score']}")
print(f"Total Sleep: {result['total_sleep_min']:.0f} min")
print(f"Deep Sleep: {result['deep_min']:.0f} min")

Example 5: Adaptive Power Management

Dynamic duty cycling based on activity state to maximize battery life.

class AdaptivePowerManager:
    """Adaptive power management for wearable devices."""

    def __init__(self):
        self.state = 'idle'
        self.sample_rates = {
            'sleep': {'imu': 1, 'ppg': 1, 'env': 0.1},
            'idle': {'imu': 10, 'ppg': 5, 'env': 0.5},
            'active': {'imu': 50, 'ppg': 25, 'env': 1},
            'workout': {'imu': 100, 'ppg': 50, 'env': 2}
        }

    def update_state(self, activity, hr):
        """Update power state based on activity and HR."""
        if activity in ['Sleeping']:
            new_state = 'sleep'
        elif activity in ['Sitting', 'Standing']:
            new_state = 'idle'
        elif activity in ['Walking']:
            new_state = 'active'
        elif activity in ['Running', 'Cycling'] or hr > 130:
            new_state = 'workout'
        else:
            new_state = 'active'

        if new_state != self.state:
            self.state = new_state
            self._apply_settings()

    def _apply_settings(self):
        """Apply sample rates to hardware."""
        rates = self.sample_rates[self.state]
        print(f"Power state: {self.state}")
        print(f"  IMU: {rates['imu']} Hz")
        print(f"  PPG: {rates['ppg']} Hz")

    def get_estimated_battery_hours(self, capacity_mah=200):
        """Estimate remaining battery life."""
        power_draw = {  # mW per state
            'sleep': 0.3,
            'idle': 1.5,
            'active': 4.0,
            'workout': 8.0
        }
        current_ma = power_draw[self.state] / 3.7  # At 3.7V
        return capacity_mah / current_ma

# Usage
power_mgr = AdaptivePowerManager()
power_mgr.update_state('Running', hr=145)
print(f"Battery life: {power_mgr.get_estimated_battery_hours():.0f}h")
