Double-EWMA Control Algorithm

The Double-EWMA (Exponentially Weighted Moving Average) control algorithm provides adaptive process control with drift detection capabilities for semiconductor manufacturing.

Mathematical Foundation

\[ \begin{align} E_1(t) &= \lambda_1 X_t + (1-\lambda_1) E_1(t-1) \\ E_2(t) &= \lambda_2 E_1(t) + (1-\lambda_2) E_2(t-1) \\ \hat{X}(t+1) &= 2E_1(t) - E_2(t) \end{align} \]

where \(\lambda_1\) and \(\lambda_2\) are smoothing parameters, and \(\hat{X}(t+1)\) is the predicted value for the next time step.

Implementation

import numpy as np

class DoubleEWMAController:
    def __init__(self, lambda1=0.3, lambda2=0.2, target=0, control_limits=3):
        self.lambda1 = lambda1
        self.lambda2 = lambda2
        self.target = target
        self.control_limits = control_limits
        
        # Initialize EWMA statistics
        self.E1 = target
        self.E2 = target
        self.prediction = target
        
        # Control limits
        self.std_dev = 1.0
        self.ucl = target + control_limits * self.std_dev
        self.lcl = target - control_limits * self.std_dev
        
    def update(self, measurement):
        """Update EWMA statistics with new measurement"""
        # Update first EWMA
        self.E1 = self.lambda1 * measurement + (1 - self.lambda1) * self.E1
        
        # Update second EWMA
        self.E2 = self.lambda2 * self.E1 + (1 - self.lambda2) * self.E2
        
        # Calculate prediction
        self.prediction = 2 * self.E1 - self.E2
        
        # Check control limits
        status = self._check_control_limits(measurement)
        
        return {
            'measurement': measurement,
            'E1': self.E1,
            'E2': self.E2,
            'prediction': self.prediction,
            'status': status
        }
    
    def _check_control_limits(self, measurement):
        """Check if measurement is within control limits"""
        if measurement > self.ucl or measurement < self.lcl:
            return 'out_of_control'
        elif abs(measurement - self.target) > 2 * self.std_dev:
            return 'warning'
        else:
            return 'in_control'

PID Controller Implementation

class PIDController:
    def __init__(self, kp=1.0, ki=0.0, kd=0.0, setpoint=0):
        self.kp = kp
        self.ki = ki
        self.kd = kd
        self.setpoint = setpoint
        
        self.previous_error = 0
        self.integral = 0
        
    def update(self, measurement, dt=1.0):
        """Calculate PID output"""
        error = self.setpoint - measurement
        
        # Proportional term
        proportional = self.kp * error
        
        # Integral term
        self.integral += error * dt
        integral = self.ki * self.integral
        
        # Derivative term
        derivative = self.kd * (error - self.previous_error) / dt
        
        # PID output
        output = proportional + integral + derivative
        
        self.previous_error = error
        
        return output

Adaptive Control

class AdaptiveController:
    def __init__(self, base_controller):
        self.base_controller = base_controller
        self.performance_history = []
        self.adaptation_rate = 0.1
        
    def adapt_parameters(self, performance_metric):
        """Adapt controller parameters based on performance"""
        self.performance_history.append(performance_metric)
        
        if len(self.performance_history) > 10:
            recent_performance = np.mean(self.performance_history[-10:])
            
            # Adjust parameters based on performance trend
            if recent_performance < 0.8:  # Poor performance
                self.base_controller.kp *= (1 + self.adaptation_rate)
                self.base_controller.ki *= (1 + self.adaptation_rate * 0.5)
            elif recent_performance > 0.95:  # Excellent performance
                self.base_controller.kp *= (1 - self.adaptation_rate * 0.5)
                self.base_controller.ki *= (1 - self.adaptation_rate * 0.5)
