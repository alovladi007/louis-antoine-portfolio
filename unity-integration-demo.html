Unity AR/VR Integration

Production-ready Unity package for real-time object recognition, pose estimation, and semantic segmentation in AR/VR applications

Object Detection

Real-time YOLO-based detection with 3D bounding boxes

Pose Estimation

33-point skeleton tracking for full body capture

Segmentation

Pixel-wise scene understanding with 150+ classes

Hand Tracking

21 keypoints per hand for gesture control

Quick Start

// Install via Unity Package Manager // Add to manifest.json: "com.arvr.recognition": "https://github.com/arvr/unity-sdk.git" // Basic usage in your MonoBehaviour using ARVR.Recognition; public class ObjectDetector : MonoBehaviour { private RecognitionEngine engine; void Start() { engine = new RecognitionEngine(); engine.Initialize(ModelType.YOLOv8); engine.OnObjectDetected += HandleDetection; } void HandleDetection(DetectionResult[] results) { foreach (var obj in results) { Debug.Log($"{obj.Label}: {obj.Confidence:P}"); SpawnBoundingBox(obj.BoundingBox3D); } } }
