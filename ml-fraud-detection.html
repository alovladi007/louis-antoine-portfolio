# Real-time fraud detection with PySpark Structured Streaming
from pyspark.sql import SparkSession
from pyspark.sql.functions import *
from pyspark.ml.feature import VectorAssembler
from sklearn.ensemble import IsolationForest

spark = SparkSession.builder \
    .appName("FraudDetection") \
    .config("spark.streaming.kafka.maxRatePerPartition", "10000") \
    .getOrCreate()

# Read from Kafka stream
df = spark.readStream \
    .format("kafka") \
    .option("kafka.bootstrap.servers", "localhost:9092") \
    .option("subscribe", "transactions") \
    .load()

# Parse and extract features
transactions = df.select(
    from_json(col("value").cast("string"), schema).alias("data")
).select("data.*")

# Apply Isolation Forest model (pre-trained)
@udf(returnType=FloatType())
def predict_anomaly(features):
    return float(model.decision_function([features])[0])

# Score transactions and flag anomalies
scored = transactions.withColumn(
    "anomaly_score",
    predict_anomaly(array("amount", "distance", "time_delta"))
).withColumn(
    "is_fraud",
    when(col("anomaly_score") < -0.5, True).otherwise(False)
)
