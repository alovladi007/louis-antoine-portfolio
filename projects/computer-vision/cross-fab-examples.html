Virtual Metrology - Basic Implementation
Complete implementation of virtual metrology using CatBoost for CD and overlay prediction. 
import pandas as pd
import numpy as np
from catboost import CatBoostRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, mean_absolute_error

class VirtualMetrology:
    def __init__(self):
        self.cd_model = None
        self.overlay_model = None
        
    def train(self, data, features, cd_target, overlay_target):
        """Train VM models for CD and overlay prediction"""
        X_train, X_test, y_cd_train, y_cd_test = train_test_split(
            data[features], data[cd_target], test_size=0.2, random_state=42
        )
        
        # Train CD model
        self.cd_model = CatBoostRegressor(
            iterations=1000,
            learning_rate=0.1,
            depth=6,
            loss_function='RMSE',
            random_seed=42
        )
        self.cd_model.fit(X_train, y_cd_train, verbose=False)
        
        # Train overlay model
        self.overlay_model = CatBoostRegressor(
            iterations=1000,
            learning_rate=0.1,
            depth=6,
            loss_function='RMSE',
            random_seed=42
        )
        self.overlay_model.fit(X_train, data[overlay_target].iloc[X_train.index], verbose=False)
        
        # Evaluate models
        cd_pred = self.cd_model.predict(X_test)
        overlay_pred = self.overlay_model.predict(X_test)
        
        print(f"CD Model RMSE: {np.sqrt(mean_squared_error(y_cd_test, cd_pred)):.3f}")
        print(f"Overlay Model RMSE: {np.sqrt(mean_squared_error(data[overlay_target].iloc[X_test.index], overlay_pred)):.3f}")
    
    def predict(self, features):
        """Predict CD and overlay for given features"""
        cd_pred = self.cd_model.predict(features)
        overlay_pred = self.overlay_model.predict(features)
        return cd_pred, overlay_pred

# Usage example
vm = VirtualMetrology()
vm.train(data, ['feature_size', 'temperature', 'pressure'], 'cd_measurement', 'overlay_measurement')
cd_pred, overlay_pred = vm.predict(new_features)

Run-to-Run Control - Double-EWMA
Implementation of double-EWMA run-to-run control with constraint handling. 
import numpy as np

class DoubleEWMAController:
    def __init__(self, target, lambda_param=0.3, control_gain=1.0, 
                 min_action=-2.0, max_action=2.0):
        self.target = target
        self.lambda_param = lambda_param
        self.control_gain = control_gain
        self.min_action = min_action
        self.max_action = max_action
        
        # Initialize EWMA states
        self.ewma1 = target
        self.ewma2 = target
        self.control_history = []
        
    def update(self, measurement):
        """Update controller with new measurement"""
        # Update EWMA states
        self.ewma1 = self.lambda_param * measurement + (1 - self.lambda_param) * self.ewma1
        self.ewma2 = self.lambda_param * self.ewma1 + (1 - self.lambda_param) * self.ewma2
        
        # Calculate control action
        control_action = -self.control_gain * (2 * self.ewma1 - self.ewma2 - self.target)
        
        # Apply constraints
        control_action = np.clip(control_action, self.min_action, self.max_action)
        
        # Store history
        self.control_history.append(control_action)
        
        return control_action
    
    def get_control_statistics(self):
        """Get control performance statistics"""
        if len(self.control_history) < 2:
            return {}
        
        actions = np.array(self.control_history)
        return {
            'mean_action': np.mean(actions),
            'std_action': np.std(actions),
            'max_action': np.max(actions),
            'min_action': np.min(actions)
        }

for measurement in measurements:
    action = controller.update(measurement)
    print(f"Measurement: {measurement:.2f}, Control Action: {action:.2f}")

SPC/FDC - Control Charts
Statistical process control implementation with T² and SPE charts. 
import numpy as np
from scipy import stats

class SPCController:
    def __init__(self, control_limit=3.0, sample_size=10):
        self.control_limit = control_limit
        self.sample_size = sample_size
        self.samples = []
        self.mean = 0
        self.cov_matrix = None
        self.inv_cov_matrix = None
        
    def add_sample(self, measurement):
        """Add new measurement and check for out-of-control conditions"""
        self.samples.append(measurement)
        
        if len(self.samples) > self.sample_size:
            self.samples.pop(0)
        
        if len(self.samples) >= self.sample_size:
            self._update_statistics()
            return self._check_control_limits(measurement)
        
        return False
    
    def _update_statistics(self):
        """Update mean and covariance matrix"""
        self.mean = np.mean(self.samples, axis=0)
        self.cov_matrix = np.cov(self.samples, rowvar=False)
        
        if self.cov_matrix.shape[0] > 0:
            try:
                self.inv_cov_matrix = np.linalg.inv(self.cov_matrix)
            except np.linalg.LinAlgError:
                self.inv_cov_matrix = np.eye(self.cov_matrix.shape[0])
    
    def _check_control_limits(self, measurement):
        """Check if measurement exceeds control limits"""
        if self.inv_cov_matrix is None:
            return False
        
        # Calculate T² statistic
        diff = measurement - self.mean
        t2 = np.dot(diff, np.dot(self.inv_cov_matrix, diff))
        
        # Check against control limit
        return t2 > self.control_limit**2
    
    def get_control_chart_data(self):
        """Get data for plotting control charts"""
        if len(self.samples) < 2:
            return None
        
        t2_values = []
        for sample in self.samples:
            diff = sample - self.mean
            t2 = np.dot(diff, np.dot(self.inv_cov_matrix, diff))
            t2_values.append(t2)
        
        return {
            't2_values': t2_values,
            'control_limit': self.control_limit**2,
            'mean': self.mean.tolist()
        }

# Usage example
spc = SPCController(control_limit=3.0, sample_size=10)
for measurement in measurements:
    is_out_of_control = spc.add_sample(measurement)
    if is_out_of_control:
        print(f"Out of control detected at measurement: {measurement}")

# Cross-Fab Complete Workflow Notebook
# This notebook demonstrates the complete Cross-Fab workflow

# 1. Data Loading and Preprocessing
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# Load process data
data = pd.read_csv('process_data.csv')
print(f"Data shape: {data.shape}")
print(f"Columns: {data.columns.tolist()}")

# 2. Exploratory Data Analysis
plt.figure(figsize=(12, 8))
sns.pairplot(data[['cd_measurement', 'overlay_measurement', 'feature_size', 'temperature']])
plt.title('Process Data Correlation Matrix')
plt.show()

# 3. Virtual Metrology Training
from cross_fab.vm import VirtualMetrology

vm = VirtualMetrology()
features = ['feature_size', 'temperature', 'pressure', 'process_time']
vm.train(data, features, 'cd_measurement', 'overlay_measurement')

# 4. Run-to-Run Control Setup
from cross_fab.r2r import DoubleEWMAController

controller = DoubleEWMAController(target=50.0)
control_actions = []
for measurement in data['cd_measurement']:
    action = controller.update(measurement)
    control_actions.append(action)

# 5. SPC/FDC Implementation
from cross_fab.spc import SPCController

spc = SPCController()
faults = []
for measurement in data[['cd_measurement', 'overlay_measurement']].values:
    is_fault = spc.add_sample(measurement)
    faults.append(is_fault)

# 6. Results Visualization
fig, axes = plt.subplots(2, 2, figsize=(15, 10))

# CD measurements and control actions
axes[0, 0].plot(data['cd_measurement'], label='CD Measurements')
axes[0, 0].plot(control_actions, label='Control Actions')
axes[0, 0].set_title('CD Control Performance')
axes[0, 0].legend()

# SPC chart
chart_data = spc.get_control_chart_data()
axes[0, 1].plot(chart_data['t2_values'], label='T² Values')
axes[0, 1].axhline(y=chart_data['control_limit'], color='r', linestyle='--', label='Control Limit')
axes[0, 1].set_title('SPC Control Chart')
axes[0, 1].legend()

# VM predictions vs actual
cd_pred, overlay_pred = vm.predict(data[features])
axes[1, 0].scatter(data['cd_measurement'], cd_pred, alpha=0.6)
axes[1, 0].plot([data['cd_measurement'].min(), data['cd_measurement'].max()], 
                [data['cd_measurement'].min(), data['cd_measurement'].max()], 'r--')
axes[1, 0].set_title('VM CD Predictions vs Actual')
axes[1, 0].set_xlabel('Actual CD')
axes[1, 0].set_ylabel('Predicted CD')

# Fault detection results
fault_indices = np.where(faults)[0]
axes[1, 1].plot(data['cd_measurement'], 'b-', label='CD Measurements')
axes[1, 1].scatter(fault_indices, data['cd_measurement'].iloc[fault_indices], 
                   color='red', s=50, label='Faults Detected')
axes[1, 1].set_title('Fault Detection Results')
axes[1, 1].legend()

plt.tight_layout()
plt.show()

print(f"Total faults detected: {sum(faults)}")
print(f"Control performance: {controller.get_control_statistics()}")
