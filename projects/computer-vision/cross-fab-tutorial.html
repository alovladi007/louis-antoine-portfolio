Virtual Metrology Setup Tutorial

1
Data Collection and Preparation
The first step in building a robust virtual metrology system is collecting and preparing high-quality process data. This involves gathering historical measurements, process parameters, and ensuring data quality.

Key Data Requirements:

  • CD Measurements: Critical dimension measurements from metrology tools (SEM, CD-SEM)
  • Overlay Data: X/Y overlay measurements from alignment systems
  • Process Parameters: Temperature, pressure, process time, gas flows
  • Wafer Information: Position, lot ID, recipe information
  • Tool Parameters: Chamber conditions, tool settings, maintenance data

Data Quality Checklist:

  • Minimum 1000 samples for reliable model training
  • Data spanning multiple lots and process conditions
  • Outlier detection and removal
  • Missing data handling strategy
  • Data validation and consistency checks
# Import required libraries
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from catboost import CatBoostRegressor

# Load and prepare data
data = pd.read_csv('process_data.csv')
features = ['feature_size', 'temperature', 'pressure', 'process_time', 'wafer_x', 'wafer_y']
targets = ['cd_measurement', 'overlay_measurement']

# Split data
X_train, X_test, y_train, y_test = train_test_split(
    data[features], data[targets], test_size=0.2, random_state=42
)
💡 Pro Tip: Use stratified sampling to ensure representative data distribution across different process conditions and wafer positions.

# Feature engineering
def create_features(df):
    # Create interaction features
    df['temp_pressure'] = df['temperature'] * df['pressure']
    df['wafer_radius'] = np.sqrt(df['wafer_x']**2 + df['wafer_y']**2)
    
    # Create lag features
    df['cd_lag1'] = df['cd_measurement'].shift(1)
    df['overlay_lag1'] = df['overlay_measurement'].shift(1)
    
    return df

# Apply feature engineering
X_train = create_features(X_train)
X_test = create_features(X_test)

# Model validation
def evaluate_model(model, X_test, y_test, target_name):
    predictions = model.predict(X_test)
    mse = np.mean((predictions - y_test)**2)
    rmse = np.sqrt(mse)
    mae = np.mean(np.abs(predictions - y_test))
    
    print(f"{target_name} Model Performance:")
    print(f"RMSE: {rmse:.3f}")
    print(f"MAE: {mae:.3f}")
    
    return predictions

Run-to-Run Control Tutorial

1
Control System Setup
Initialize the run-to-run control system with appropriate parameters. 
class R2RController:
    def __init__(self, target_cd, ewma_lambda=0.3, control_gain=1.0):
        self.target_cd = target_cd
        self.ewma_lambda = ewma_lambda
        self.control_gain = control_gain
        self.ewma1 = target_cd
        self.ewma2 = target_cd
        self.control_history = []
        
    def update(self, measured_cd):
        # Double-EWMA calculation
        self.ewma1 = self.ewma_lambda * measured_cd + (1 - self.ewma_lambda) * self.ewma1
        self.ewma2 = self.ewma_lambda * self.ewma1 + (1 - self.ewma_lambda) * self.ewma2
        
        # Control action calculation
        control_action = -self.control_gain * (2 * self.ewma1 - self.ewma2 - self.target_cd)
        
        # Apply constraints
        control_action = np.clip(control_action, -2.0, 2.0)
        
        self.control_history.append(control_action)
        return control_action

Kalman Filter Implementation
Implement Kalman filtering for advanced state estimation and control. 
class KalmanR2RController:
    def __init__(self, target_cd, process_noise=0.1, measurement_noise=0.05):
        self.target_cd = target_cd
        self.process_noise = process_noise
        self.measurement_noise = measurement_noise
        self.state = target_cd
        self.covariance = 1.0
        self.control_gain = 1.0
        
    def update(self, measured_cd):
        # Prediction step
        predicted_state = self.state
        predicted_covariance = self.covariance + self.process_noise
        
        # Update step
        kalman_gain = predicted_covariance / (predicted_covariance + self.measurement_noise)
        self.state = predicted_state + kalman_gain * (measured_cd - predicted_state)
        self.covariance = (1 - kalman_gain) * predicted_covariance
        
        # Control action
        control_action = -self.control_gain * (self.state - self.target_cd)
        control_action = np.clip(control_action, -2.0, 2.0)
        
        return control_action

Control Chart Setup
Configure statistical process control charts for real-time monitoring. 
class SPCController:
    def __init__(self, control_limit=3.0, sample_size=10):
        self.control_limit = control_limit
        self.sample_size = sample_size
        self.samples = []
        self.mean = 0
        self.std = 0
        
    def add_sample(self, value):
        self.samples.append(value)
        if len(self.samples) > self.sample_size:
            self.samples.pop(0)
            
        # Update statistics
        self.mean = np.mean(self.samples)
        self.std = np.std(self.samples, ddof=1)
        
        # Check for out-of-control conditions
        if len(self.samples) >= self.sample_size:
            z_score = abs(value - self.mean) / self.std if self.std > 0 else 0
            return z_score > self.control_limit
        
        return False

Fault Detection System
Implement fault detection and classification system. 
class FaultDetector:
    def __init__(self):
        self.fault_patterns = {
            'trend': self.detect_trend,
            'shift': self.detect_shift,
            'cycle': self.detect_cycle
        }
        
    def detect_trend(self, samples):
        if len(samples) < 7:
            return False
        # Check for 7 consecutive points trending up or down
        recent = samples[-7:]
        return all(recent[i] < recent[i+1] for i in range(6)) or \
               all(recent[i] > recent[i+1] for i in range(6))
    
    def detect_shift(self, samples):
        if len(samples) < 8:
            return False
        # Check for 8 consecutive points on one side of center line
        recent = samples[-8:]
        center = np.mean(samples[:-8]) if len(samples) > 8 else np.mean(samples)
        return all(x > center for x in recent) or all(x < center for x in recent)
    
    def detect_cycle(self, samples):
        if len(samples) < 14:
            return False
        # Simple cycle detection using autocorrelation
        recent = samples[-14:]
        autocorr = np.correlate(recent, recent, mode='full')
        return np.max(autocorr) > 0.8 * len(recent)
