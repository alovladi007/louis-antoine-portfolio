1. Backtesting Framework Architecture

Comprehensive backtesting framework designed specifically for reinforcement learning trading systems, providing realistic market simulation, performance evaluation, and strategy validation.

Framework Components:

Core Backtesting Engine

import pandas as pd import numpy as np from datetime import datetime, timedelta import matplotlib.pyplot as plt import seaborn as sns class BacktestingEngine: def __init__(self, initial_capital=100000, transaction_cost=0.001, slippage=0.0005): self.initial_capital = initial_capital self.transaction_cost = transaction_cost self.slippage = slippage # Results storage self.trades = [] self.portfolio_values = [] self.returns = [] self.dates = [] def run_backtest(self, strategy, data, start_date=None, end_date=None): """Run complete backtest for given strategy and data""" # Prepare data if start_date: data = data[data.index >= start_date] if end_date: data = data[data.index <= end_date] # Initialize portfolio portfolio = Portfolio(self.initial_capital) # Reset results self._reset_results() # Run backtest for i, (date, row) in enumerate(data.iterrows()): current_price = row['close'] # Get strategy decision state = self._get_state(data, i) action = strategy.predict(state) # Execute trade trade_result = self._execute_trade(portfolio, action, current_price, date) # Record results self._record_results(portfolio, trade_result, date) # Calculate final metrics metrics = self._calculate_metrics() return BacktestResults(metrics, self.trades, self.portfolio_values) def _execute_trade(self, portfolio, action, price, date): """Execute trading action with realistic costs""" trade_result = {'date': date, 'price': price, 'action': 'hold', 'shares': 0, 'cost': 0} if action == 0: # Buy shares_to_buy = portfolio.cash // (price * (1 + self.transaction_cost + self.slippage)) if shares_to_buy > 0: cost = shares_to_buy * price * (1 + self.transaction_cost + self.slippage) portfolio.cash -= cost portfolio.shares += shares_to_buy portfolio.avg_cost = (portfolio.avg_cost * (portfolio.shares - shares_to_buy) + cost) / portfolio.shares trade_result.update({ 'action': 'buy', 'shares': shares_to_buy, 'cost': cost }) elif action == 2: # Sell if portfolio.shares > 0: revenue = portfolio.shares * price * (1 - self.transaction_cost - self.slippage) portfolio.cash += revenue trade_result.update({ 'action': 'sell', 'shares': portfolio.shares, 'cost': revenue }) portfolio.shares = 0 portfolio.avg_cost = 0 return trade_result def _get_state(self, data, index, lookback=20): """Get market state for strategy decision""" start_idx = max(0, index - lookback) state_data = data.iloc[start_idx:index + 1] # Calculate technical indicators features = self._calculate_features(state_data) return features def _calculate_features(self, data): """Calculate technical features for state representation""" features = {} # Price features features['price'] = data['close'].iloc[-1] features['returns'] = data['close'].pct_change().iloc[-1] features['volatility'] = data['close'].pct_change().std() # Technical indicators features['sma_20'] = data['close'].rolling(20).mean().iloc[-1] features['rsi'] = self._calculate_rsi(data['close']) features['macd'] = self._calculate_macd(data['close']) # Volume features features['volume'] = data['volume'].iloc[-1] features['volume_ratio'] = data['volume'].iloc[-1] / data['volume'].mean() return np.array(list(features.values())) def _calculate_rsi(self, prices, period=14): """Calculate RSI indicator""" if len(prices) < period + 1: return 50 delta = prices.diff() gain = (delta.where(delta > 0, 0)).rolling(window=period).mean() loss = (-delta.where(delta < 0, 0)).rolling(window=period).mean() rs = gain / loss rsi = 100 - (100 / (1 + rs)) return rsi.iloc[-1] if not pd.isna(rsi.iloc[-1]) else 50 def _calculate_macd(self, prices, fast=12, slow=26): """Calculate MACD indicator""" if len(prices) < slow: return 0 ema_fast = prices.ewm(span=fast).mean() ema_slow = prices.ewm(span=slow).mean() macd_line = ema_fast - ema_slow return macd_line.iloc[-1] if not pd.isna(macd_line.iloc[-1]) else 0

class PerformanceMetrics: def __init__(self, portfolio_values, returns, trades, risk_free_rate=0.02): self.portfolio_values = np.array(portfolio_values) self.returns = np.array(returns) self.trades = trades self.risk_free_rate = risk_free_rate def calculate_all_metrics(self): """Calculate comprehensive performance metrics""" metrics = {} # Return metrics metrics.update(self._calculate_return_metrics()) # Risk metrics metrics.update(self._calculate_risk_metrics()) # Trading metrics metrics.update(self._calculate_trading_metrics()) # Advanced metrics metrics.update(self._calculate_advanced_metrics()) return metrics def _calculate_return_metrics(self): """Calculate return-based metrics""" total_return = (self.portfolio_values[-1] - self.portfolio_values[0]) / self.portfolio_values[0] annual_return = (1 + total_return) ** (252 / len(self.returns)) - 1 # Cumulative returns cumulative_returns = self.portfolio_values / self.portfolio_values[0] - 1 return { 'total_return': total_return, 'annual_return': annual_return, 'cumulative_return': cumulative_returns[-1], 'average_daily_return': np.mean(self.returns), 'median_daily_return': np.median(self.returns) } def _calculate_risk_metrics(self): """Calculate risk-based metrics""" # Volatility daily_volatility = np.std(self.returns) annual_volatility = daily_volatility * np.sqrt(252) # Sharpe ratio excess_returns = self.returns - self.risk_free_rate / 252 sharpe_ratio = np.mean(excess_returns) / np.std(excess_returns) * np.sqrt(252) # Sortino ratio downside_returns = self.returns[self.returns < 0] downside_deviation = np.std(downside_returns) * np.sqrt(252) sortino_ratio = (np.mean(self.returns) * 252 - self.risk_free_rate) / downside_deviation # Maximum drawdown max_drawdown = self._calculate_max_drawdown() # VaR and CVaR var_95 = np.percentile(self.returns, 5) cvar_95 = np.mean(self.returns[self.returns <= var_95]) return { 'volatility': annual_volatility, 'sharpe_ratio': sharpe_ratio, 'sortino_ratio': sortino_ratio, 'max_drawdown': max_drawdown, 'var_95': var_95, 'cvar_95': cvar_95, 'downside_deviation': downside_deviation } def _calculate_trading_metrics(self): """Calculate trading-specific metrics""" if not self.trades: return {'num_trades': 0, 'win_rate': 0, 'avg_trade': 0} buy_trades = [t for t in self.trades if t['action'] == 'buy'] sell_trades = [t for t in self.trades if t['action'] == 'sell'] # Trade analysis trade_returns = [] for i, sell_trade in enumerate(sell_trades): # Find corresponding buy trade buy_trade = None for buy in buy_trades: if buy['date'] < sell_trade['date']: buy_trade = buy if buy_trade: trade_return = (sell_trade['cost'] - buy_trade['cost']) / buy_trade['cost'] trade_returns.append(trade_return) win_rate = np.mean([r > 0 for r in trade_returns]) if trade_returns else 0 avg_trade_return = np.mean(trade_returns) if trade_returns else 0 # Turnover total_turnover = sum([t['cost'] for t in self.trades]) avg_portfolio_value = np.mean(self.portfolio_values) turnover_ratio = total_turnover / avg_portfolio_value return { 'num_trades': len(self.trades), 'win_rate': win_rate, 'avg_trade_return': avg_trade_return, 'total_turnover': total_turnover, 'turnover_ratio': turnover_ratio, 'num_buy_trades': len(buy_trades), 'num_sell_trades': len(sell_trades) } def _calculate_advanced_metrics(self): """Calculate advanced performance metrics""" # Calmar ratio max_drawdown = self._calculate_max_drawdown() annual_return = (1 + (self.portfolio_values[-1] - self.portfolio_values[0]) / self.portfolio_values[0]) ** (252 / len(self.returns)) - 1 calmar_ratio = annual_return / abs(max_drawdown) if max_drawdown != 0 else 0 # Information ratio (vs benchmark) # Assuming benchmark return of 10% annually benchmark_return = 0.10 / 252 excess_returns = self.returns - benchmark_return tracking_error = np.std(excess_returns) * np.sqrt(252) information_ratio = np.mean(excess_returns) * np.sqrt(252) / tracking_error if tracking_error != 0 else 0 # Beta calculation (simplified) # In practice, this would be calculated against a market benchmark beta = 1.0 # Placeholder - would need benchmark data # Alpha calculation alpha = annual_return - (self.risk_free_rate + beta * (0.10 - self.risk_free_rate)) return { 'calmar_ratio': calmar_ratio, 'information_ratio': information_ratio, 'beta': beta, 'alpha': alpha, 'tracking_error': tracking_error } def _calculate_max_drawdown(self): """Calculate maximum drawdown""" peak = np.maximum.accumulate(self.portfolio_values) drawdown = (self.portfolio_values - peak) / peak return np.min(drawdown)

def calculate_risk_adjusted_metrics(returns, portfolio_values): """Calculate comprehensive risk-adjusted performance metrics""" # Tail risk metrics tail_ratio = calculate_tail_ratio(returns) common_sense_ratio = calculate_common_sense_ratio(returns) # Regime-based metrics regime_metrics = calculate_regime_metrics(returns) # Stability metrics stability_metrics = calculate_stability_metrics(portfolio_values) return { 'tail_ratio': tail_ratio, 'common_sense_ratio': common_sense_ratio, 'regime_metrics': regime_metrics, 'stability_metrics': stability_metrics } def calculate_tail_ratio(returns): """Calculate tail ratio (95th percentile / 5th percentile)""" p95 = np.percentile(returns, 95) p5 = np.percentile(returns, 5) return abs(p95 / p5) if p5 != 0 else 0 def calculate_common_sense_ratio(returns): """Calculate common sense ratio (average positive return / average negative return)""" positive_returns = returns[returns > 0] negative_returns = returns[returns < 0] if len(positive_returns) == 0 or len(negative_returns) == 0: return 0 avg_positive = np.mean(positive_returns) avg_negative = abs(np.mean(negative_returns)) return avg_positive / avg_negative if avg_negative != 0 else 0

3. Strategy Validation

Walk-Forward Analysis

class WalkForwardAnalysis: def __init__(self, training_period=252, testing_period=63, step_size=21): self.training_period = training_period self.testing_period = testing_period self.step_size = step_size def run_analysis(self, strategy, data): """Run walk-forward analysis""" results = [] start_date = data.index[0] end_date = data.index[-1] current_start = start_date while current_start + timedelta(days=self.training_period + self.testing_period) <= end_date: # Define periods train_start = current_start train_end = current_start + timedelta(days=self.training_period) test_start = train_end test_end = test_start + timedelta(days=self.testing_period) # Get training and testing data train_data = data[(data.index >= train_start) & (data.index < train_end)] test_data = data[(data.index >= test_start) & (data.index < test_end)] if len(train_data) < self.training_period * 0.8 or len(test_data) < self.testing_period * 0.8: current_start += timedelta(days=self.step_size) continue # Train strategy strategy.train(train_data) # Test strategy backtest_engine = BacktestingEngine() test_results = backtest_engine.run_backtest(strategy, test_data) # Store results results.append({ 'train_start': train_start, 'train_end': train_end, 'test_start': test_start, 'test_end': test_end, 'metrics': test_results.metrics }) current_start += timedelta(days=self.step_size) return WalkForwardResults(results) def analyze_stability(self, walk_forward_results): """Analyze stability of strategy performance""" metrics_list = [result['metrics'] for result in walk_forward_results.results] stability_analysis = {} # Key metrics to analyze key_metrics = ['annual_return', 'sharpe_ratio', 'max_drawdown', 'win_rate'] for metric in key_metrics: values = [m[metric] for m in metrics_list if metric in m] if values: stability_analysis[metric] = { 'mean': np.mean(values), 'std': np.std(values), 'min': np.min(values), 'max': np.max(values), 'stability_ratio': np.mean(values) / np.std(values) if np.std(values) != 0 else 0 } return stability_analysis

class MonteCarloAnalysis: def __init__(self, num_simulations=1000, confidence_level=0.95): self.num_simulations = num_simulations self.confidence_level = confidence_level def run_simulation(self, returns, strategy_func): """Run Monte Carlo simulation""" simulation_results = [] for _ in range(self.num_simulations): # Bootstrap sample returns bootstrap_returns = np.random.choice(returns, size=len(returns), replace=True) # Calculate metrics for bootstrap sample metrics = self._calculate_bootstrap_metrics(bootstrap_returns) simulation_results.append(metrics) # Calculate confidence intervals confidence_intervals = self._calculate_confidence_intervals(simulation_results) return MonteCarloResults(simulation_results, confidence_intervals) def _calculate_bootstrap_metrics(self, returns): """Calculate metrics for bootstrap sample""" # Simulate portfolio values portfolio_values = [100000] for ret in returns: portfolio_values.append(portfolio_values[-1] * (1 + ret)) # Calculate key metrics total_return = (portfolio_values[-1] - portfolio_values[0]) / portfolio_values[0] volatility = np.std(returns) * np.sqrt(252) sharpe_ratio = np.mean(returns) / np.std(returns) * np.sqrt(252) if np.std(returns) != 0 else 0 # Maximum drawdown peak = np.maximum.accumulate(portfolio_values) drawdown = (np.array(portfolio_values) - peak) / peak max_drawdown = np.min(drawdown) return { 'total_return': total_return, 'volatility': volatility, 'sharpe_ratio': sharpe_ratio, 'max_drawdown': max_drawdown } def _calculate_confidence_intervals(self, simulation_results): """Calculate confidence intervals for metrics""" alpha = 1 - self.confidence_level lower_percentile = (alpha / 2) * 100 upper_percentile = (1 - alpha / 2) * 100 confidence_intervals = {} # Calculate intervals for each metric for metric in simulation_results[0].keys(): values = [result[metric] for result in simulation_results] confidence_intervals[metric] = { 'lower': np.percentile(values, lower_percentile), 'upper': np.percentile(values, upper_percentile), 'mean': np.mean(values) } return confidence_intervals
