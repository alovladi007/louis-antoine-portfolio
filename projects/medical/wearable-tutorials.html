Tutorial 1: Reading IMU Data

Beginner

Learn to read and visualize accelerometer and gyroscope data from a 6-axis IMU.

import numpy as np
from wearable.sensors import IMU

# Initialize IMU at 50 Hz
imu = IMU(sample_rate=50)

# Read sensor data
accel = imu.read_accelerometer()  # [ax, ay, az] in m/s²
gyro = imu.read_gyroscope()       # [gx, gy, gz] in rad/s

# Compute magnitude
accel_mag = np.linalg.norm(accel)
print(f"Acceleration: {accel_mag:.2f} m/s²")

# Detect orientation from gravity
pitch = np.arctan2(accel[0], np.sqrt(accel[1]**2 + accel[2]**2))
roll = np.arctan2(accel[1], accel[2])
print(f"Pitch: {np.degrees(pitch):.1f}°, Roll: {np.degrees(roll):.1f}°")

Tutorial 2: Madgwick Sensor Fusion

Intermediate

Implement the Madgwick filter to combine accelerometer and gyroscope for drift-free orientation.

from wearable import SensorFusion
import matplotlib.pyplot as plt

# Initialize Madgwick filter
fusion = SensorFusion(
    sample_rate=50,
    filter_type='madgwick',
    beta=0.1  # Higher = more accelerometer trust
)

euler_history = []

# Process 10 seconds of data
for i in range(500):
    accel, gyro = imu.read()

    # Update filter
    fusion.update(accel, gyro)

    # Get Euler angles
    roll, pitch, yaw = fusion.get_euler()
    euler_history.append([roll, pitch, yaw])

# Plot orientation
euler_history = np.array(euler_history)
plt.plot(euler_history[:, 0], label='Roll')
plt.plot(euler_history[:, 1], label='Pitch')
plt.plot(euler_history[:, 2], label='Yaw')
plt.legend()
plt.show()

Tutorial 3: PPG Heart Rate Estimation

Intermediate

Extract heart rate from photoplethysmography (PPG) signals using peak detection.

from scipy import signal
import numpy as np

def estimate_heart_rate(ppg, sample_rate=25):
    # Bandpass filter: 0.5-4 Hz (30-240 BPM)
    sos = signal.butter(4, [0.5, 4], btype='band',
                        fs=sample_rate, output='sos')
    filtered = signal.sosfilt(sos, ppg)

    # Find peaks (systolic)
    peaks, _ = signal.find_peaks(
        filtered,
        distance=sample_rate * 0.4,  # min 0.4s between beats
        prominence=np.std(filtered) * 0.5
    )

    # Calculate RR intervals
    rr_intervals = np.diff(peaks) / sample_rate  # seconds

    # Heart rate from mean RR
    if len(rr_intervals) > 0:
        hr = 60 / np.mean(rr_intervals)
        return hr, peaks, rr_intervals
    return None, peaks, []

# Usage
hr, peaks, rr = estimate_heart_rate(ppg_signal)
print(f"Heart Rate: {hr:.0f} BPM")

Tutorial 4: Activity Recognition with CNN

Advanced

Train a 1D CNN to classify activities from accelerometer windows.

import tensorflow as tf
from tensorflow.keras import layers, models

def build_activity_cnn(input_shape=(128, 6), num_classes=8):
    model = models.Sequential([
        # Conv block 1
        layers.Conv1D(64, 5, activation='relu', input_shape=input_shape),
        layers.BatchNormalization(),
        layers.MaxPooling1D(2),

        # Conv block 2
        layers.Conv1D(128, 5, activation='relu'),
        layers.BatchNormalization(),
        layers.MaxPooling1D(2),

        # Conv block 3
        layers.Conv1D(256, 3, activation='relu'),
        layers.GlobalAveragePooling1D(),

        # Classifier
        layers.Dense(128, activation='relu'),
        layers.Dropout(0.5),
        layers.Dense(num_classes, activation='softmax')
    ])

    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    return model

Tutorial 5: Sleep Stage Classification

Advanced

Classify sleep stages using movement and heart rate features.

from wearable import SleepTracker
import numpy as np

def extract_sleep_features(accel_epoch, hr_epoch):
    """Extract features from 30-second epoch."""
    # Activity count (integrated acceleration)
    activity = np.sum(np.abs(np.diff(accel_epoch)))

    # HR statistics
    hr_mean = np.mean(hr_epoch)
    hr_std = np.std(hr_epoch)

    # Movement intensity
    accel_var = np.var(accel_epoch, axis=0).sum()

    return [activity, hr_mean, hr_std, accel_var]

def classify_sleep_stage(features, model):
    """Classify epoch into sleep stage."""
    # 0: Awake, 1: REM, 2: Light, 3: Deep
    pred = model.predict([features])
    stages = ['Awake', 'REM', 'Light', 'Deep']
    return stages[np.argmax(pred)]

Tutorial 6: Quantizing for Edge Deployment

Expert

Convert a trained model to int8 for microcontroller deployment.

import tensorflow as tf

def quantize_model(model, calibration_data):
    """Convert float32 model to int8 TFLite."""

    # Representative dataset for calibration
    def representative_dataset():
        for sample in calibration_data:
            yield [sample.astype(np.float32)]

    # Convert with full integer quantization
    converter = tf.lite.TFLiteConverter.from_keras_model(model)
    converter.optimizations = [tf.lite.Optimize.DEFAULT]
    converter.representative_dataset = representative_dataset
    converter.target_spec.supported_ops = [
        tf.lite.OpsSet.TFLITE_BUILTINS_INT8
    ]
    converter.inference_input_type = tf.int8
    converter.inference_output_type = tf.int8

    tflite_model = converter.convert()

    # Save quantized model
    with open('model_int8.tflite', 'wb') as f:
        f.write(tflite_model)

    print(f"Model size: {len(tflite_model) / 1024:.1f} KB")
    return tflite_model

# Quantize trained activity model
quantize_model(activity_model, X_train[:100])

Tutorial 7: Real-Time Health Monitoring Pipeline

Expert

Build a complete real-time pipeline combining all sensors and ML models.

from wearable import (
    SensorFusion, ActivityRecognizer,
    HeartRateMonitor, AlertSystem
)
import asyncio

class HealthMonitor:
    def __init__(self):
        self.fusion = SensorFusion(sample_rate=50)
        self.activity = ActivityRecognizer('activity.tflite')
        self.hr_monitor = HeartRateMonitor(sample_rate=25)
        self.alerts = AlertSystem()

        self.accel_buffer = []
        self.ppg_buffer = []

    async def process_sensors(self, accel, gyro, ppg):
        # Update orientation
        self.fusion.update(accel, gyro)
        euler = self.fusion.get_euler()

        # Buffer for activity recognition
        self.accel_buffer.append(np.concatenate([accel, gyro]))
        if len(self.accel_buffer) >= 128:
            window = np.array(self.accel_buffer[-128:])
            activity, conf = self.activity.classify(window)
            self.accel_buffer = self.accel_buffer[-64:]  # 50% overlap

        # Heart rate estimation
        self.ppg_buffer.append(ppg)
        if len(self.ppg_buffer) >= 125:  # 5 seconds
            hr = self.hr_monitor.estimate(
                np.array(self.ppg_buffer),
                np.linalg.norm(accel)  # motion ref
            )
            self.ppg_buffer = self.ppg_buffer[-62:]

            # Check alerts
            self.alerts.check_heart_rate(hr)

        return {
            'orientation': euler,
            'activity': activity,
            'heart_rate': hr
        }

# Run monitor
monitor = HealthMonitor()
while True:
    data = await monitor.process_sensors(accel, gyro, ppg)
