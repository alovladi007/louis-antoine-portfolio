2. Basic Signal Filtering

Apply bandpass and notch filters to remove noise and powerline interference.

20 min 
from scipy.signal import butter, filtfilt, iirnotch

def bandpass_filter(signal, low, high, fs, order=4):
    """Butterworth bandpass filter"""
    nyq = fs / 2
    b, a = butter(order, [low/nyq, high/nyq], btype='band')
    return filtfilt(b, a, signal)

def notch_filter(signal, freq, fs, q=30):
    """Remove powerline interference"""
    b, a = iirnotch(freq, q, fs)
    return filtfilt(b, a, signal)

# Apply filters to ECG
filtered = bandpass_filter(signal, 0.5, 40, fs)
filtered = notch_filter(filtered, 60, fs)  # 60 Hz for US

4. Heart Rate Variability Analysis

Compute time-domain and frequency-domain HRV metrics for autonomic assessment.

30 min 
import numpy as np
from scipy.signal import welch

def compute_hrv(rr_intervals):
    """Compute HRV metrics from RR intervals (ms)"""
    rr = np.array(rr_intervals)
    diff_rr = np.diff(rr)

    # Time domain
    sdnn = np.std(rr)
    rmssd = np.sqrt(np.mean(diff_rr**2))
    pnn50 = np.sum(np.abs(diff_rr) > 50) / len(diff_rr) * 100

    # Frequency domain (interpolate to 4 Hz)
    fs_hrv = 4
    t_rr = np.cumsum(rr) / 1000
    t_interp = np.arange(0, t_rr[-1], 1/fs_hrv)
    rr_interp = np.interp(t_interp, t_rr, rr)

    f, psd = welch(rr_interp - np.mean(rr_interp), fs_hrv, nperseg=256)

    lf = np.trapz(psd[(f >= 0.04) & (f < 0.15)])
    hf = np.trapz(psd[(f >= 0.15) & (f < 0.4)])

    return {'sdnn': sdnn, 'rmssd': rmssd, 'pnn50': pnn50,
            'lf': lf, 'hf': hf, 'lf_hf': lf/hf}

5. EEG Band Power Extraction

Extract and analyze power in delta, theta, alpha, beta, and gamma bands.

25 min 
from scipy.signal import welch

def extract_band_power(eeg, fs):
    """Extract EEG band powers using Welch's method"""
    bands = {
        'delta': (0.5, 4),
        'theta': (4, 8),
        'alpha': (8, 13),
        'beta': (13, 30),
        'gamma': (30, 100)
    }

    f, psd = welch(eeg, fs, nperseg=fs*2)
    total_power = np.trapz(psd[(f >= 0.5) & (f <= 100)])

    powers = {}
    for name, (low, high) in bands.items():
        mask = (f >= low) & (f < high)
        powers[name] = np.trapz(psd[mask])
        powers[f'{name}_rel'] = powers[name] / total_power * 100

    return powers

# Example: Compute alpha/beta ratio (attention index)
powers = extract_band_power(eeg_signal, 256)
attention = powers['beta'] / powers['alpha']

6. EMG Fatigue Analysis

Track muscle fatigue through median frequency decline during contraction.

30 min 
def analyze_fatigue(emg, fs, window_s=1.0):
    """Track median frequency over time for fatigue analysis"""
    window = int(window_s * fs)
    step = window // 2  # 50% overlap

    mdf_values = []
    time_stamps = []

    for i in range(0, len(emg) - window, step):
        segment = emg[i:i+window]
        f, psd = welch(segment, fs, nperseg=window//2)

        # Median frequency: frequency that divides PSD area in half
        cumsum = np.cumsum(psd)
        mdf_idx = np.searchsorted(cumsum, cumsum[-1]/2)
        mdf = f[mdf_idx]

        mdf_values.append(mdf)
        time_stamps.append((i + window/2) / fs)

    # Calculate fatigue index (slope of MDF decline)
    slope, _ = np.polyfit(time_stamps, mdf_values, 1)
    fatigue_index = -slope / mdf_values[0] * 100  # % decline/second

    return time_stamps, mdf_values, fatigue_index

7. Artifact Removal with ICA

Use Independent Component Analysis to remove eye blink artifacts from EEG.

35 min 
from sklearn.decomposition import FastICA
import numpy as np

def remove_artifacts_ica(eeg_multichannel, n_components=None):
    """Remove artifacts using FastICA"""
    # eeg_multichannel: shape (channels, samples)

    ica = FastICA(n_components=n_components, random_state=42)
    sources = ica.fit_transform(eeg_multichannel.T).T  # Independent components

    # Identify artifact components (high kurtosis for blinks)
    from scipy.stats import kurtosis
    kurt_values = [kurtosis(s) for s in sources]
    artifact_idx = [i for i, k in enumerate(kurt_values) if k > 5]

    # Zero out artifact components
    sources_clean = sources.copy()
    sources_clean[artifact_idx] = 0

    # Reconstruct clean signal
    eeg_clean = ica.inverse_transform(sources_clean.T).T

    return eeg_clean, artifact_idx

8. Building an Arrhythmia Classifier

Train a CNN to classify ECG beats into Normal, AFib, PVC, and other arrhythmias.

45 min 
import tensorflow as tf
from tensorflow.keras import layers, models

def build_ecg_cnn(input_length=360, n_classes=5):
    """1D CNN for ECG beat classification"""
    model = models.Sequential([
        layers.Input(shape=(input_length, 1)),

        # Conv Block 1
        layers.Conv1D(32, 5, padding='same'),
        layers.BatchNormalization(),
        layers.ReLU(),
        layers.MaxPooling1D(2),

        # Conv Block 2
        layers.Conv1D(64, 5, padding='same'),
        layers.BatchNormalization(),
        layers.ReLU(),
        layers.MaxPooling1D(2),

        # Conv Block 3
        layers.Conv1D(128, 3, padding='same'),
        layers.BatchNormalization(),
        layers.ReLU(),
        layers.GlobalAveragePooling1D(),

        # Classifier
        layers.Dense(64, activation='relu'),
        layers.Dropout(0.5),
        layers.Dense(n_classes, activation='softmax')
    ])

    model.compile(optimizer='adam',
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])
    return model

def build_sleep_lstm(n_features=15, seq_length=20, n_classes=5):
    """LSTM for sequence-to-sequence sleep staging"""
    model = models.Sequential([
        layers.Input(shape=(seq_length, n_features)),

        # Bidirectional LSTM layers
        layers.Bidirectional(layers.LSTM(64, return_sequences=True)),
        layers.Dropout(0.3),
        layers.Bidirectional(layers.LSTM(32)),
        layers.Dropout(0.3),

        # Classification head
        layers.Dense(32, activation='relu'),
        layers.Dense(n_classes, activation='softmax')
    ])

    return model

10. Real-Time Signal Processing

Implement streaming analysis with circular buffers and online filtering.

40 min 
from scipy.signal import lfilter, lfilter_zi

class RealtimeProcessor:
    def __init__(self, fs, buffer_size=1024):
        self.fs = fs
        self.buffer = np.zeros(buffer_size)
        self.idx = 0

        # Initialize filter state
        self.b, self.a = butter(4, [0.5/180, 40/180], btype='band')
        self.zi = lfilter_zi(self.b, self.a)

    def process_sample(self, sample):
        """Process single sample in real-time"""
        # Online filtering
        filtered, self.zi = lfilter(self.b, self.a, [sample], zi=self.zi)

        # Update circular buffer
        self.buffer[self.idx % len(self.buffer)] = filtered[0]
        self.idx += 1

        # Detect R-peak if enough samples
        if self.idx >= 100:
            recent = self._get_recent(100)
            if recent[-1] > np.mean(recent) + 2*np.std(recent):
                return True  # R-peak detected
        return False
