"""
Complete ECG Analysis Pipeline
Processes raw ECG → Filtering → R-peaks → HRV → Arrhythmia Detection → Report
"""

import numpy as np
import wfdb
from scipy.signal import butter, filtfilt, iirnotch
from biosppy.signals import ecg
import tensorflow as tf
from datetime import datetime


class ECGAnalyzer:
    """Complete ECG analysis with arrhythmia detection"""

    ARRHYTHMIA_CLASSES = ['Normal', 'AFib', 'PVC', 'PAC', 'VT']

    def __init__(self, fs=360, model_path=None):
        self.fs = fs
        self.model = tf.keras.models.load_model(model_path) if model_path else None

    def preprocess(self, signal):
        """Apply bandpass and notch filtering"""
        # Bandpass 0.5-40 Hz
        nyq = self.fs / 2
        b, a = butter(4, [0.5/nyq, 40/nyq], btype='band')
        filtered = filtfilt(b, a, signal)

        # Notch filter 60 Hz (US) or 50 Hz (EU)
        b_notch, a_notch = iirnotch(60, 30, self.fs)
        filtered = filtfilt(b_notch, a_notch, filtered)

        return filtered

    def detect_r_peaks(self, signal):
        """Detect R-peaks using biosppy"""
        out = ecg.ecg(signal=signal, sampling_rate=self.fs, show=False)
        return out['rpeaks']

    def compute_hrv(self, r_peaks):
        """Calculate HRV metrics"""
        rr = np.diff(r_peaks) / self.fs * 1000  # ms
        diff_rr = np.diff(rr)

        return {
            'mean_hr': 60000 / np.mean(rr),
            'sdnn': np.std(rr),
            'rmssd': np.sqrt(np.mean(diff_rr**2)),
            'pnn50': np.sum(np.abs(diff_rr) > 50) / len(diff_rr) * 100,
            'rr_intervals': rr.tolist()
        }

    def extract_beats(self, signal, r_peaks, window=180):
        """Extract individual beats around R-peaks"""
        beats = []
        for r in r_peaks:
            if r - window//2 >= 0 and r + window//2 < len(signal):
                beat = signal[r - window//2:r + window//2]
                # Normalize
                beat = (beat - np.mean(beat)) / (np.std(beat) + 1e-8)
                beats.append(beat)
        return np.array(beats)

    def classify_beats(self, beats):
        """Classify beats using trained model"""
        if self.model is None:
            return ['Normal'] * len(beats)

        predictions = self.model.predict(beats[..., np.newaxis], verbose=0)
        labels = [self.ARRHYTHMIA_CLASSES[np.argmax(p)] for p in predictions]
        return labels

    def analyze(self, signal):
        """Complete analysis pipeline"""
        # Preprocess
        filtered = self.preprocess(signal)

        # Detect R-peaks
        r_peaks = self.detect_r_peaks(filtered)

        # Compute HRV
        hrv = self.compute_hrv(r_peaks)

        # Extract and classify beats
        beats = self.extract_beats(filtered, r_peaks)
        classifications = self.classify_beats(beats)

        # Count arrhythmias
        arrhythmia_counts = {}
        for cls in self.ARRHYTHMIA_CLASSES:
            arrhythmia_counts[cls] = classifications.count(cls)

        return {
            'filtered_signal': filtered,
            'r_peaks': r_peaks,
            'hrv': hrv,
            'beat_classifications': classifications,
            'arrhythmia_summary': arrhythmia_counts,
            'total_beats': len(beats)
        }

    def generate_report(self, results):
        """Generate clinical summary report"""
        report = f"""
========================================
ECG Analysis Report - {datetime.now().strftime('%Y-%m-%d %H:%M')}
========================================

HEART RATE ANALYSIS
  Mean Heart Rate: {results['hrv']['mean_hr']:.1f} BPM
  Total Beats: {results['total_beats']}

HRV METRICS (Time Domain)
  SDNN:  {results['hrv']['sdnn']:.1f} ms
  RMSSD: {results['hrv']['rmssd']:.1f} ms
  pNN50: {results['hrv']['pnn50']:.1f}%

ARRHYTHMIA DETECTION
"""
        for cls, count in results['arrhythmia_summary'].items():
            pct = count / results['total_beats'] * 100
            report += f"  {cls}: {count} ({pct:.1f}%)\n"

        return report


# Usage example
if __name__ == '__main__':
    # Load MIT-BIH data
    record = wfdb.rdrecord('mit-bih/100', pn_dir='mitdb')
    signal = record.p_signal[:, 0]

    # Analyze
    analyzer = ECGAnalyzer(fs=360)
    results = analyzer.analyze(signal)

    # Print report
    print(analyzer.generate_report(results))

"""
Multi-Channel EEG Processor
Handles artifact removal, spectral analysis, and state classification
"""

import numpy as np
from scipy.signal import butter, filtfilt, welch
from sklearn.decomposition import FastICA
from scipy.stats import kurtosis


class EEGProcessor:
    """Multi-channel EEG processing with ICA artifact removal"""

    BANDS = {
        'delta': (0.5, 4),
        'theta': (4, 8),
        'alpha': (8, 13),
        'beta': (13, 30),
        'gamma': (30, 100)
    }

    STATES = ['Alert', 'Relaxed', 'Drowsy', 'Asleep']

    def __init__(self, fs=256, channels=None):
        self.fs = fs
        self.channels = channels or ['Fp1', 'Fp2', 'C3', 'C4', 'O1', 'O2']

    def bandpass_filter(self, data, low, high):
        """Apply bandpass filter to all channels"""
        nyq = self.fs / 2
        b, a = butter(4, [low/nyq, high/nyq], btype='band')
        return filtfilt(b, a, data, axis=1)

    def remove_artifacts_ica(self, data, n_components=None):
        """Remove artifacts using ICA"""
        n_components = n_components or data.shape[0]

        ica = FastICA(n_components=n_components, random_state=42)
        sources = ica.fit_transform(data.T).T

        # Detect artifact components via kurtosis
        artifact_idx = []
        for i, src in enumerate(sources):
            k = kurtosis(src)
            # High kurtosis = blink, very low = muscle
            if k > 5 or k < -1:
                artifact_idx.append(i)

        # Zero out artifacts and reconstruct
        sources_clean = sources.copy()
        sources_clean[artifact_idx] = 0

        clean_data = ica.inverse_transform(sources_clean.T).T

        return clean_data, artifact_idx

    def compute_band_powers(self, data, epoch_s=2):
        """Compute power in each frequency band"""
        nperseg = epoch_s * self.fs

        powers = {band: [] for band in self.BANDS}

        for ch_data in data:
            f, psd = welch(ch_data, self.fs, nperseg=nperseg)
            total = np.trapz(psd[(f >= 0.5) & (f <= 100)])

            for band, (low, high) in self.BANDS.items():
                mask = (f >= low) & (f < high)
                band_power = np.trapz(psd[mask])
                powers[band].append(band_power / total * 100)  # Relative %

        # Average across channels
        return {band: np.mean(vals) for band, vals in powers.items()}

    def classify_state(self, band_powers):
        """Classify brain state from band powers"""
        alpha = band_powers['alpha']
        beta = band_powers['beta']
        theta = band_powers['theta']
        delta = band_powers['delta']

        # Simple rule-based classification
        if delta > 40:
            return 'Asleep'
        elif theta > 30:
            return 'Drowsy'
        elif alpha > 25 and beta < 20:
            return 'Relaxed'
        else:
            return 'Alert'

    def compute_asymmetry(self, data):
        """Compute frontal alpha asymmetry (FAA)"""
        # Assuming Fp1=0, Fp2=1
        f, psd_left = welch(data[0], self.fs)
        _, psd_right = welch(data[1], self.fs)

        alpha_mask = (f >= 8) & (f < 13)
        alpha_left = np.log(np.trapz(psd_left[alpha_mask]))
        alpha_right = np.log(np.trapz(psd_right[alpha_mask]))

        # Positive FAA = more left activation (approach motivation)
        faa = alpha_right - alpha_left
        return faa

    def process(self, raw_data):
        """Complete EEG processing pipeline"""
        # Filter 0.5-100 Hz
        filtered = self.bandpass_filter(raw_data, 0.5, 100)

        # Remove artifacts
        clean, artifacts = self.remove_artifacts_ica(filtered)

        # Compute band powers
        powers = self.compute_band_powers(clean)

        # Classify state
        state = self.classify_state(powers)

        # Compute asymmetry
        faa = self.compute_asymmetry(clean)

        return {
            'clean_data': clean,
            'removed_components': len(artifacts),
            'band_powers': powers,
            'brain_state': state,
            'frontal_asymmetry': faa
        }


# Usage
processor = EEGProcessor(fs=256, channels=['Fp1', 'Fp2', 'C3', 'C4', 'O1', 'O2'])
results = processor.process(eeg_data)  # shape: (6, n_samples)
print(f"Brain State: {results['brain_state']}")
print(f"Alpha Power: {results['band_powers']['alpha']:.1f}%")

"""
EMG Gesture Recognition
Feature extraction and real-time classification for prosthetic control
"""

import numpy as np
from scipy.signal import butter, filtfilt, welch
from sklearn.ensemble import RandomForestClassifier
from collections import deque


class EMGGestureRecognizer:
    """Real-time EMG gesture classification"""

    GESTURES = ['Rest', 'Fist', 'Open', 'Pinch', 'Point', 'Wrist Flex']

    def __init__(self, fs=1000, n_channels=8, window_ms=200):
        self.fs = fs
        self.n_channels = n_channels
        self.window = int(window_ms * fs / 1000)
        self.model = RandomForestClassifier(n_estimators=100)
        self.is_trained = False

        # Real-time buffer
        self.buffer = deque(maxlen=self.window)

    def preprocess(self, data):
        """Bandpass filter 20-500 Hz"""
        nyq = self.fs / 2
        b, a = butter(4, [20/nyq, 450/nyq], btype='band')
        return filtfilt(b, a, data, axis=1)

    def extract_features(self, segment):
        """Extract time and frequency domain features"""
        features = []

        for ch in range(segment.shape[0]):
            ch_data = segment[ch]

            # Time domain
            mav = np.mean(np.abs(ch_data))
            rms = np.sqrt(np.mean(ch_data**2))
            wl = np.sum(np.abs(np.diff(ch_data)))
            zc = np.sum(np.diff(np.sign(ch_data)) != 0)
            ssc = np.sum(np.diff(np.sign(np.diff(ch_data))) != 0)

            # Frequency domain
            f, psd = welch(ch_data, self.fs, nperseg=len(ch_data)//2)
            mdf_idx = np.searchsorted(np.cumsum(psd), np.sum(psd)/2)
            mdf = f[min(mdf_idx, len(f)-1)]
            mnf = np.sum(f * psd) / np.sum(psd)

            features.extend([mav, rms, wl, zc, ssc, mdf, mnf])

        return np.array(features)

    def train(self, X_train, y_train):
        """Train gesture classifier"""
        # X_train: list of segments (n_channels, n_samples)
        # y_train: gesture labels

        features = []
        for segment in X_train:
            processed = self.preprocess(segment)
            feat = self.extract_features(processed)
            features.append(feat)

        X = np.array(features)
        self.model.fit(X, y_train)
        self.is_trained = True

        return self.model.score(X, y_train)

    def predict(self, segment):
        """Classify single gesture"""
        if not self.is_trained:
            raise ValueError("Model not trained")

        processed = self.preprocess(segment)
        features = self.extract_features(processed).reshape(1, -1)

        pred_idx = self.model.predict(features)[0]
        proba = self.model.predict_proba(features)[0]

        return self.GESTURES[pred_idx], proba[pred_idx]

    def process_realtime(self, sample):
        """Process streaming samples"""
        self.buffer.append(sample)

        if len(self.buffer) == self.window:
            segment = np.array(self.buffer).T  # (n_channels, window)
            gesture, confidence = self.predict(segment)

            if confidence > 0.7:  # Confidence threshold
                return gesture, confidence

        return None, 0

    def analyze_fatigue(self, data):
        """Track muscle fatigue via MDF decline"""
        window_s = 1.0
        window = int(window_s * self.fs)
        step = window // 2

        mdf_over_time = []

        for i in range(0, data.shape[1] - window, step):
            segment = data[:, i:i+window]

            # Average MDF across channels
            mdfs = []
            for ch in range(segment.shape[0]):
                f, psd = welch(segment[ch], self.fs)
                mdf_idx = np.searchsorted(np.cumsum(psd), np.sum(psd)/2)
                mdfs.append(f[min(mdf_idx, len(f)-1)])

            mdf_over_time.append(np.mean(mdfs))

        # Calculate fatigue rate (MDF slope)
        time_points = np.arange(len(mdf_over_time)) * step / self.fs
        slope, _ = np.polyfit(time_points, mdf_over_time, 1)

        fatigue_rate = -slope / mdf_over_time[0] * 100  # %/second

        return {
            'mdf_timeline': mdf_over_time,
            'fatigue_rate': fatigue_rate,
            'initial_mdf': mdf_over_time[0],
            'final_mdf': mdf_over_time[-1]
        }

"""
Seizure Detection with LSTM
Patient-specific model for epileptic seizure onset detection
"""

import tensorflow as tf
from tensorflow.keras import layers, models
import numpy as np


def build_seizure_detector(n_channels=23, seq_length=256, n_features=8):
    """LSTM model for seizure detection"""

    # Per-channel feature input (seq_length, n_channels * n_features)
    inputs = layers.Input(shape=(seq_length, n_channels * n_features))

    # Bidirectional LSTM layers
    x = layers.Bidirectional(layers.LSTM(64, return_sequences=True))(inputs)
    x = layers.Dropout(0.3)(x)
    x = layers.Bidirectional(layers.LSTM(32))(x)
    x = layers.Dropout(0.3)(x)

    # Attention mechanism
    attention = layers.Dense(1, activation='tanh')(x)
    attention = layers.Flatten()(attention)
    attention = layers.Activation('softmax')(attention)

    # Classification head
    x = layers.Dense(32, activation='relu')(x)
    outputs = layers.Dense(1, activation='sigmoid')(x)

    model = models.Model(inputs, outputs)
    model.compile(
        optimizer=tf.keras.optimizers.Adam(lr=0.001),
        loss='binary_crossentropy',
        metrics=['accuracy', tf.keras.metrics.AUC(name='auc')]
    )

    return model


class SeizureDetector:
    """Real-time seizure detection system"""

    def __init__(self, model_path, fs=256, window_s=2, threshold=0.8):
        self.model = tf.keras.models.load_model(model_path)
        self.fs = fs
        self.window = window_s * fs
        self.threshold = threshold
        self.alert_cooldown = 30 * fs  # 30s cooldown
        self.last_alert = -self.alert_cooldown

    def extract_features(self, epoch):
        """Extract features from EEG epoch"""
        features = []
        for ch in range(epoch.shape[0]):
            ch_data = epoch[ch]
            # Statistical + spectral features
            f = [
                np.mean(ch_data), np.std(ch_data),
                np.max(ch_data) - np.min(ch_data),
                np.sum(np.abs(np.diff(ch_data))),
                # ... band powers, entropy, etc.
            ]
            features.extend(f)
        return np.array(features)

    def detect(self, eeg_window, sample_idx):
        """Detect seizure in window"""
        features = self.extract_features(eeg_window)
        prob = self.model.predict(features.reshape(1, -1), verbose=0)[0, 0]

        is_seizure = prob > self.threshold

        # Apply cooldown to prevent repeated alerts
        if is_seizure and (sample_idx - self.last_alert) > self.alert_cooldown:
            self.last_alert = sample_idx
            return {'seizure': True, 'probability': prob, 'alert': True}

        return {'seizure': is_seizure, 'probability': prob, 'alert': False}

Automatic Sleep Staging

AASM-compliant 5-stage sleep classification from polysomnography data.

"""
Automatic Sleep Staging (AASM Standard)
Classifies 30-second epochs into Wake, N1, N2, N3, REM
"""

import numpy as np
from scipy.signal import welch
import tensorflow as tf


class SleepStager:
    """AASM-compliant sleep stage classifier"""

    STAGES = ['Wake', 'N1', 'N2', 'N3', 'REM']
    EPOCH_S = 30  # AASM standard

    def __init__(self, fs=100, model_path=None):
        self.fs = fs
        self.epoch_samples = self.EPOCH_S * fs

        if model_path:
            self.model = tf.keras.models.load_model(model_path)
        else:
            self.model = self._build_model()

    def _build_model(self, n_features=20, seq_context=11):
        """Build sequence-to-sequence sleep staging model"""
        model = tf.keras.Sequential([
            tf.keras.layers.Input(shape=(seq_context, n_features)),
            tf.keras.layers.Bidirectional(tf.keras.layers.LSTM(64, return_sequences=True)),
            tf.keras.layers.Dropout(0.3),
            tf.keras.layers.Bidirectional(tf.keras.layers.LSTM(32)),
            tf.keras.layers.Dense(32, activation='relu'),
            tf.keras.layers.Dense(5, activation='softmax')
        ])
        model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])
        return model

    def extract_epoch_features(self, eeg, eog=None, emg=None):
        """Extract features from 30s epoch"""
        features = []

        # EEG band powers
        f, psd = welch(eeg, self.fs, nperseg=self.fs*4)
        total = np.trapz(psd[(f >= 0.5) & (f <= 50)])

        bands = [(0.5, 4), (4, 8), (8, 13), (13, 30), (11, 16)]  # Last = spindle
        for low, high in bands:
            mask = (f >= low) & (f < high)
            features.append(np.trapz(psd[mask]) / total)

        # Statistical features
        features.extend([np.std(eeg), np.max(eeg) - np.min(eeg)])

        # EOG activity (eye movements)
        if eog is not None:
            features.append(np.std(eog))
            features.append(np.sum(np.abs(np.diff(eog))))

        # EMG activity (chin tone)
        if emg is not None:
            features.append(np.sqrt(np.mean(emg**2)))

        return np.array(features)

    def stage_night(self, eeg_signal, eog=None, emg=None):
        """Stage entire night recording"""
        n_epochs = len(eeg_signal) // self.epoch_samples

        # Extract features for all epochs
        all_features = []
        for i in range(n_epochs):
            start = i * self.epoch_samples
            end = start + self.epoch_samples

            eeg_epoch = eeg_signal[start:end]
            eog_epoch = eog[start:end] if eog is not None else None
            emg_epoch = emg[start:end] if emg is not None else None

            features = self.extract_epoch_features(eeg_epoch, eog_epoch, emg_epoch)
            all_features.append(features)

        # Create context windows (±5 epochs)
        stages = []
        for i in range(n_epochs):
            context = []
            for j in range(i-5, i+6):
                if 0 <= j < n_epochs:
                    context.append(all_features[j])
                else:
                    context.append(np.zeros_like(all_features[0]))

            pred = self.model.predict(np.array([context]), verbose=0)
            stages.append(self.STAGES[np.argmax(pred)])

        return stages

    def generate_hypnogram(self, stages):
        """Generate sleep statistics"""
        counts = {s: stages.count(s) for s in self.STAGES}
        total = len(stages) * 0.5  # minutes

        return {
            'total_sleep_time': (total - counts['Wake'] * 0.5),
            'sleep_efficiency': (1 - counts['Wake'] / len(stages)) * 100,
            'n3_percent': counts['N3'] / len(stages) * 100,
            'rem_percent': counts['REM'] / len(stages) * 100,
            'stage_counts': counts
        }
