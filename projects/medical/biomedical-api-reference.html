API Reference

Python API documentation for biomedical signal processing

class ECGProcessor

Core Module

ECG signal analysis including R-peak detection, HRV analysis, and arrhythmia classification.

from biosignal import ECGProcessor

ecg = ECGProcessor(sample_rate=360)
ecg.load_signal(data)
peaks = ecg.detect_r_peaks(method="pan_tompkins")
hrv = ecg.compute_hrv()
print(f"Heart Rate: {hrv['mean_hr']:.1f} BPM")
print(f"SDNN: {hrv['sdnn']:.1f} ms")
detect_r_peaks(method="pan_tompkins", threshold=0.5)

Detect R-peaks using Pan-Tompkins, Hamilton, or wavelet methods.

compute_hrv(time_domain=True, freq_domain=True)

Calculate HRV metrics: SDNN, RMSSD, pNN50, LF/HF ratio.

classify_arrhythmia(model="cnn")

Classify beats into Normal, AFib, VT, VF, PVC categories.

class EEGProcessor

EEG Module

EEG signal processing with band power extraction and sleep staging.

from biosignal import EEGProcessor

eeg = EEGProcessor(sample_rate=256, channels=['Fp1', 'C3', 'O1'])
eeg.load_signal(data)
eeg.apply_bandpass(0.5, 50)

bands = eeg.extract_band_power()
print(f"Alpha power: {bands['alpha']:.2f} μV²")
extract_band_power(bands=["delta", "theta", "alpha", "beta", "gamma"])

Extract absolute and relative power in frequency bands.

detect_seizures(model="lstm", threshold=0.8)

Detect seizure events with onset timestamps.

classify_sleep_stage(epoch_length=30)

Classify into Wake, N1, N2, N3, REM stages.

class EMGProcessor

EMG Module 
from biosignal import EMGProcessor

emg = EMGProcessor(sample_rate=1000)
emg.load_signal(data)
emg.apply_bandpass(20, 500)

envelope = emg.compute_envelope(window_ms=50)
fatigue = emg.analyze_fatigue()
print(f"Median freq shift: {fatigue['mdf_slope']:.2f} Hz/s")
compute_envelope(window_ms=50, method="rms")

Compute RMS or MAV envelope for activation detection.

analyze_fatigue(window_s=1.0)

Track median frequency decline for fatigue assessment.

classify_gesture(model="cnn")

Recognize hand gestures from forearm EMG.

class SignalFilter

Filtering 
from biosignal.filters import SignalFilter

filt = SignalFilter(sample_rate=360)

# ECG preprocessing
filtered = filt.bandpass(data, low=0.5, high=40, order=4)
filtered = filt.notch(filtered, freq=50, q=30)

# Artifact removal
clean = filt.remove_baseline(filtered, cutoff=0.5)
bandpass(data, low, high, order=4, method="butter")
notch(data, freq, q=30)
wavelet_denoise(data, wavelet="db4", level=4)

class FeatureExtractor

Features 
from biosignal.features import FeatureExtractor

fe = FeatureExtractor()
time_feats = fe.time_domain(signal)  # mean, std, rms, skew, kurt
freq_feats = fe.frequency_domain(signal, fs=256)
wavelet_feats = fe.wavelet_features(signal, wavelet="db4")

# Combine all features
features = fe.extract_all(signal, fs=256)
print(f"Feature vector: {len(features)} dimensions")
time_domain(signal) → mean, std, rms, skewness, kurtosis, zero_crossings
frequency_domain(signal, fs) → peak_freq, mean_freq, median_freq, spectral_entropy
wavelet_features(signal, wavelet="db4", level=5) → approximation and detail coefficients statistics
