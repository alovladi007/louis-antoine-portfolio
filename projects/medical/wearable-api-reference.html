class SensorFusion

Core Module

9-axis IMU sensor fusion using Madgwick or Kalman filters for orientation estimation.

from wearable import SensorFusion

fusion = SensorFusion(
    sample_rate=50,
    filter_type='madgwick',
    beta=0.1
)

# Process sensor data
for accel, gyro, mag in sensor_stream:
    quaternion = fusion.update(accel, gyro, mag)
    euler = fusion.get_euler()  # roll, pitch, yaw
    print(f"Orientation: {euler}")
update(accel, gyro, mag=None)

Update orientation estimate with new sensor readings. Returns quaternion [w, x, y, z].

get_euler()

Get current orientation as Euler angles (roll, pitch, yaw) in degrees.

get_linear_acceleration()

Returns gravity-compensated linear acceleration in world frame.

reset()

Reset filter state to initial quaternion [1, 0, 0, 0].

class ActivityRecognizer

ML Module

CNN-based human activity recognition from accelerometer and gyroscope data.

from wearable import ActivityRecognizer

recognizer = ActivityRecognizer(
    model_path='models/activity_cnn.tflite',
    window_size=128,  # 2.56s at 50Hz
    num_classes=8
)

# Real-time classification
activity, confidence = recognizer.classify(sensor_window)
print(f"Activity: {activity} ({confidence:.1%})")

# Available activities
# 0: Walking, 1: Running, 2: Sitting, 3: Standing
# 4: Stairs Up, 5: Stairs Down, 6: Cycling, 7: Sleeping
classify(window)

Classify activity from sensor window (N, 6). Returns (activity_id, confidence).

get_probabilities(window)

Returns probability distribution over all activity classes.

extract_features(window)

Extract time and frequency domain features for classical ML.

class HeartRateMonitor

PPG Module

PPG-based heart rate and HRV estimation with motion artifact removal.

from wearable import HeartRateMonitor

hr_monitor = HeartRateMonitor(
    sample_rate=25,
    use_motion_compensation=True
)

# Process PPG with accelerometer for motion compensation
hr = hr_monitor.estimate(ppg_signal, accel_signal)
hrv = hr_monitor.compute_hrv(duration=300)  # 5-minute HRV

print(f"HR: {hr} BPM")
print(f"RMSSD: {hrv['rmssd']:.1f} ms")
print(f"LF/HF: {hrv['lf_hf_ratio']:.2f}")
estimate(ppg, accel=None)

Estimate instantaneous heart rate from PPG signal. Optional accelerometer for motion compensation.

compute_hrv(duration=300)

Compute HRV metrics over specified duration. Returns dict with SDNN, RMSSD, pNN50, LF, HF.

detect_peaks()

Detect systolic peaks in PPG. Returns peak indices and RR intervals.

get_signal_quality()

Returns signal quality index (0-1) based on SNR and artifact detection.

class SleepTracker

Sleep Module 
from wearable import SleepTracker

tracker = SleepTracker(epoch_length=30)  # 30-second epochs

# Analyze overnight data
sleep_data = tracker.analyze(
    accel_data,
    hr_data,
    start_time='23:00',
    end_time='07:00'
)

print(f"Total sleep: {sleep_data['total_sleep_min']} min")
print(f"Sleep efficiency: {sleep_data['efficiency']:.1%}")
print(f"Deep sleep: {sleep_data['deep_min']} min")
analyze(accel, hr, start_time, end_time)

Perform full sleep analysis. Returns dict with stages, duration, efficiency, and score.

classify_epoch(accel_epoch, hr_epoch)

Classify single epoch as Awake, REM, Light, or Deep sleep.

get_hypnogram()

Returns sleep stage sequence for visualization.

class EdgeProcessor

TinyML Module 
from wearable import EdgeProcessor

# Initialize processor with model
processor = EdgeProcessor(
    model_path='models/activity_int8.tflite',
    arena_size=48 * 1024  # 48 KB tensor arena
)

# Run inference
output = processor.invoke(input_data)
latency = processor.get_latency_ms()

print(f"Inference: {latency:.1f} ms")
print(f"Memory: {processor.get_memory_usage()} bytes")
invoke(input_data)

Run inference on input tensor. Handles quantization/dequantization automatically.

get_latency_ms()

Returns inference latency of last invocation in milliseconds.

quantize_model(model_path, calibration_data)

Convert float32 model to int8 using post-training quantization.

benchmark(iterations=100)

Run benchmark and return latency statistics (mean, std, min, max).
