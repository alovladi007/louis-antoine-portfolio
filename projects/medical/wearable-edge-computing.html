TinyML Deployment (TensorFlow Lite Micro)

#include "tensorflow/lite/micro/micro_interpreter.h"
#include "activity_model.h"

// Allocate tensor arena (48KB for activity model)
constexpr int kTensorArenaSize = 48 * 1024;
uint8_t tensor_arena[kTensorArenaSize];

void setup_model() {
    // Load quantized model
    model = tflite::GetModel(activity_model_data);

    // Build interpreter with only needed ops
    static tflite::MicroMutableOpResolver<6> resolver;
    resolver.AddConv2D();
    resolver.AddMaxPool2D();
    resolver.AddFullyConnected();
    resolver.AddReshape();
    resolver.AddSoftmax();
    resolver.AddQuantize();

    interpreter = new tflite::MicroInterpreter(
        model, resolver, tensor_arena, kTensorArenaSize);
    interpreter->AllocateTensors();
}

int classify_activity(float* accel_data, int window_size) {
    // Quantize input: float -> int8
    int8_t* input = interpreter->input(0)->data.int8;
    for (int i = 0; i < window_size * 3; i++) {
        input[i] = (int8_t)(accel_data[i] * 127.0f / 4.0f);
    }

    // Run inference (~12ms on Cortex-M4 @ 64MHz)
    interpreter->Invoke();

    // Get predicted class
    int8_t* output = interpreter->output(0)->data.int8;
    int max_idx = 0;
    for (int i = 1; i < NUM_CLASSES; i++) {
        if (output[i] > output[max_idx]) max_idx = i;
    }
    return max_idx;
}
