Quick Start

Python 
from climate_forecaster import HybridForecaster

# Initialize the hybrid Prophet + LSTM forecaster
forecaster = HybridForecaster(
    prophet_config={"yearly_seasonality": True},
    lstm_config={"units": 128, "sequence_length": 168}
)

# Fit on historical data
forecaster.fit(
    df=historical_data,
    target_col="energy_demand",
    weather_cols=["temperature", "humidity", "cloud_cover"]
)

# Generate 7-day forecast with uncertainty
forecast = forecaster.predict(horizon=168, return_conf_int=True)
print(f"Next 24h demand: {forecast.iloc[:24]['yhat'].sum():.0f} MWh")
