Core Functions

DCDC_Init()

Initialize the DC-DC converter peripherals and control loop.

void DCDC_Init(DCDC_Config_t *config);

// Example:
DCDC_Config_t config = {
    .vin_nominal = 12.0f,
    .vout_target = 1.2f,
    .fsw_khz = 500,
    .current_limit = 20.0f
};
DCDC_Init(&config);

DCDC_SetVoltage()

Set the output voltage setpoint.

HAL_StatusTypeDef DCDC_SetVoltage(float voltage_v);

// Example:
DCDC_SetVoltage(1.8f);  // Set output to 1.8V

DCDC_GetStatus()

Read current converter status and measurements.

typedef struct {
    float vin;
    float vout;
    float iout;
    float temperature;
    uint8_t fault_flags;
    float efficiency;
} DCDC_Status_t;

DCDC_Status_t status;
DCDC_GetStatus(&status);

Protection

Fault Flags

#define FAULT_NONE          0x00
#define FAULT_OVP           0x01  // Over-voltage
#define FAULT_UVP           0x02  // Under-voltage
#define FAULT_OCP           0x04  // Over-current
#define FAULT_OTP           0x08  // Over-temperature
#define FAULT_UVLO          0x10  // Under-voltage lockout

DCDC_SetProtection()

void DCDC_SetProtection(DCDC_Protection_t *prot);

DCDC_Protection_t prot = {
    .ovp_threshold = 1.35f,   // 112.5% of Vout
    .uvp_threshold = 1.05f,   // 87.5% of Vout
    .ocp_threshold = 25.0f,   // Amps
    .otp_threshold = 100.0f   // Celsius
};
DCDC_SetProtection(&prot);

Callbacks

// Register fault callback
void DCDC_RegisterFaultCallback(void (*callback)(uint8_t faults));

// Register power good callback
void DCDC_RegisterPGoodCallback(void (*callback)(bool pgood));

// Example usage:
void OnFault(uint8_t faults) {
    if (faults & FAULT_OCP) {
        LED_SetRed();
        Log("Over-current detected!");
    }
}
DCDC_RegisterFaultCallback(OnFault);
