Project Overview

Why Samsung Cares

Memory bandwidth/W is a boardroom metric; SI/PI + thermals directly impact yield, RAS, and performance throttling in AI servers and mobile devices. This project delivers a comprehensive co-design methodology that optimizes signal integrity, power integrity, and thermal management simultaneously.

Technical Approach

  • Channel Modeling: TSV stack + interposer + package + DIMM trace (SPICE + S-parameters)
  • PDN Design: Z(f) shaping and target impedance vs. transient current analysis
  • Thermal Loop: Compact thermal netlists (RC) coupled to eye margin via temp-dependent jitter/noise
  • Firmware Governor: Bandwidth/latency/temperature triage with adaptive control

@dataclass
class ChannelParameters:
    """HBM channel parameters for SI modeling"""
    # TSV parameters
    tsv_height: float = 50e-6  # 50 μm
    tsv_diameter: float = 5e-6  # 5 μm
    tsv_pitch: float = 15e-6   # 15 μm
    tsv_count: int = 1024      # Per stack
    
    # Interposer parameters
    trace_width: float = 2e-6   # 2 μm
    trace_thickness: float = 1e-6  # 1 μm
    dielectric_height: float = 2e-6  # 2 μm
    dielectric_dk: float = 3.9
    loss_tangent: float = 0.02
    
    # Package parameters
    pkg_trace_width: float = 25e-6   # 25 μm
    pkg_trace_length: float = 5e-3   # 5 mm
    pkg_dk: float = 4.2
    pkg_loss_tan: float = 0.015

class HBMChannelModel:
    """Complete HBM channel model including TSV, interposer, and package"""
    
    def __init__(self, params: ChannelParameters):
        self.params = params
        self.frequency = np.logspace(6, 12, 1000)  # 1 MHz to 1 THz
        
    def calculate_tsv_impedance(self) -> np.ndarray:
        """Calculate TSV impedance using cylindrical transmission line model"""
        # Characteristic impedance of TSV
        d = self.params.tsv_diameter
        h = self.params.tsv_height
        
        # Simplified model for TSV in silicon substrate (εr ≈ 11.9)
        Z0_tsv = 60 / np.sqrt(11.9) * np.log(4 * h / d)
        
        # Frequency-dependent effects
        f = self.frequency
        
        # Skin effect resistance
        sigma_cu = 5.8e7  # Copper conductivity
        mu0 = 4 * np.pi * 1e-7
        delta = 1 / np.sqrt(np.pi * f * mu0 * sigma_cu)  # Skin depth
        
        R_dc = 4 * h / (np.pi * d**2 * sigma_cu)
        R_ac = R_dc * d / (4 * delta) * (1 + 1j)
        
        # Inductance (nH)
        L_tsv = mu0 * h / (2 * np.pi) * (np.log(4 * h / d) - 1) * 1e9
        
        # Capacitance (pF) 
        eps0 = 8.854e-12
        eps_si = 11.9
        C_tsv = 2 * np.pi * eps0 * eps_si * h / np.log(4 * h / d) * 1e12
        
        # Complex impedance
        s = 2j * np.pi * f
        Z_tsv = R_ac + s * L_tsv * 1e-9 + 1 / (s * C_tsv * 1e-12)
        
        return Z_tsv
    
    def calculate_interposer_model(self) -> rf.Network:
        """Generate interposer S-parameter model"""
        f = self.frequency
        
        # Microstrip line parameters
        w = self.params.trace_width
        h = self.params.dielectric_height
        t = self.params.trace_thickness
        er = self.params.dielectric_dk
        tand = self.params.loss_tangent
        
        # Characteristic impedance (Wheeler's formula)
        w_eff = w + t * (1 + 1/er) / np.pi * np.log(2 * np.e / t * h)
        Z0 = 60 / np.sqrt((er + 1)/2 + (er - 1)/2 * (1 + 12*h/w_eff)**(-0.5))
        
        # Effective permittivity
        er_eff = (er + 1)/2 + (er - 1)/2 * (1 + 12*h/w)**(-0.5)
        
        # Propagation constant
        c = 3e8
        beta = 2 * np.pi * f * np.sqrt(er_eff) / c
        
        # Attenuation (conductor + dielectric losses)
        # Conductor loss
        Rs = np.sqrt(np.pi * f * 4e-7 * np.pi / 5.8e7)  # Surface resistance
        alpha_c = Rs / (Z0 * w) * (1 + 2*h/w) * 8.686  # dB/m
        
        # Dielectric loss
        alpha_d = np.pi * f * np.sqrt(er_eff) * tand / c * 8.686  # dB/m
        
        alpha_total = (alpha_c + alpha_d) / 8.686  # Np/m
        
        # Propagation constant
        gamma = alpha_total + 1j * beta
        
        # Length (assume 2mm interposer routing)
        length = 2e-3
        
        # S-parameters for transmission line
        Z_line = Z0
        gamma_l = gamma * length
        
        S11 = np.zeros_like(f, dtype=complex)
        S21 = np.exp(-gamma_l)
        S12 = S21
        S22 = S11
        
        # Create network
        s_params = np.zeros((len(f), 2, 2), dtype=complex)
        s_params[:, 0, 0] = S11
        s_params[:, 0, 1] = S12
        s_params[:, 1, 0] = S21
        s_params[:, 1, 1] = S22
        
        interposer = rf.Network(frequency=rf.Frequency.from_f(f, unit='Hz'),
                               s=s_params, z0=50)
        
        return interposer
    
    def calculate_package_model(self) -> rf.Network:
        """Generate package S-parameter model"""
        f = self.frequency
        
        # Package trace parameters
        w = self.params.pkg_trace_width
        length = self.params.pkg_trace_length
        er = self.params.pkg_dk
        tand = self.params.pkg_loss_tan
        
        # Simplified stripline model
        Z0_pkg = 50  # Typical package impedance
        er_eff = er
        
        # Propagation constant
        c = 3e8
        beta = 2 * np.pi * f * np.sqrt(er_eff) / c
        alpha = np.pi * f * np.sqrt(er_eff) * tand / c
        
        gamma = alpha + 1j * beta
        gamma_l = gamma * length
        
        # S-parameters
        S11 = np.zeros_like(f, dtype=complex)
        S21 = np.exp(-gamma_l)
        S12 = S21
        S22 = S11
        
        s_params = np.zeros((len(f), 2, 2), dtype=complex)
        s_params[:, 0, 0] = S11
        s_params[:, 0, 1] = S12
        s_params[:, 1, 0] = S21
        s_params[:, 1, 1] = S22
        
        package = rf.Network(frequency=rf.Frequency.from_f(f, unit='Hz'),
                            s=s_params, z0=50)
        
        return package
    
    def generate_complete_channel(self) -> rf.Network:
        """Generate complete channel model (TSV + Interposer + Package)"""
        
        # Get individual components
        interposer = self.calculate_interposer_model()
        package = self.calculate_package_model()
        
        # TSV as impedance discontinuity
        Z_tsv = self.calculate_tsv_impedance()
        
        # Convert TSV impedance to S-parameters
        Z0 = 50
        gamma = (Z_tsv - Z0) / (Z_tsv + Z0)
        
        tsv_s11 = gamma
        tsv_s21 = 1 + gamma
        tsv_s12 = tsv_s21
        tsv_s22 = gamma
        
        s_params_tsv = np.zeros((len(self.frequency), 2, 2), dtype=complex)
        s_params_tsv[:, 0, 0] = tsv_s11
        s_params_tsv[:, 0, 1] = tsv_s12
        s_params_tsv[:, 1, 0] = tsv_s21
        s_params_tsv[:, 1, 1] = tsv_s22
        
        tsv_network = rf.Network(frequency=rf.Frequency.from_f(self.frequency, unit='Hz'),
                                s=s_params_tsv, z0=50)
        
        # Cascade all components
        complete_channel = tsv_network ** interposer ** package
        
        return complete_channel
    
    def analyze_eye_diagram(self, data_rate: float = 12.8e9) -> Dict:

    def calculate_target_impedance(self, voltage_ripple_percent: float = 5) -> float:
        """Calculate PDN target impedance"""
        
        # Target impedance = Voltage ripple / Peak current
        voltage_ripple = self.vdd_core * voltage_ripple_percent / 100
        Z_target = voltage_ripple / self.idd_peak
        
        return Z_target
    
    def design_decoupling_network(self, Z_target: float) -> Dict:
        """Design multi-stage decoupling capacitor network"""
        
        # Decoupling capacitor stages
        decap_stages = [
            {'name': 'Bulk', 'C': 470e-6, 'ESR': 0.005, 'ESL': 0.5e-9, 'count': 4},
            {'name': 'Mid', 'C': 10e-6, 'ESR': 0.01, 'ESL': 0.3e-9, 'count': 20},

            {'name': 'Ultra-high', 'C': 1e-9, 'ESR': 0.1, 'ESL': 0.1e-9, 'count': 500}
        ]
        
        f = self.frequency
        s = 2j * np.pi * f
        
        # Calculate total PDN impedance
        Z_total = np.zeros_like(f, dtype=complex)
        
        for stage in decap_stages:
            C = stage['C']
            ESR = stage['ESR']
            ESL = stage['ESL']
            count = stage['count']
            
            # Single capacitor impedance (RLC model)
            Z_cap = ESR + s * ESL + 1 / (s * C)
            
            # Parallel combination of 'count' capacitors
            Z_stage = Z_cap / count
            
            # Add to total (parallel combination)
            if np.all(Z_total == 0):
                Z_total = Z_stage
            else:
                Z_total = 1 / (1/Z_total + 1/Z_stage)
        
        # Add VRM impedance (simplified)

            'frequency_hz': f,
            'impedance_ohm': Z_total,

            'decap_stages': decap_stages
        }

        # Current step function
        i_step = np.zeros_like(t)
        step_time = 1e-9  # 1 ns rise time
        i_step[t > step_time] = current_step
        
        # Apply rise time
        rise_indices = (t > 0) & (t <= step_time)
        i_step[rise_indices] = current_step * t[rise_indices] / step_time
        
        # Convert PDN to time domain (simplified)
        # In practice, would use SPICE simulation
        
        # Simplified RC response

        # Voltage droop calculation
        v_droop = np.zeros_like(t)
        for i, time in enumerate(t):
            if time > step_time:

        # Find maximum droop and settling time

        settling_time = np.where(v_droop >= 0.9 * max_droop)[0]
        settling_time = t[settling_time[0]] if len(settling_time) > 0 else t[-1]
        
        return {
            'time_ns': t * 1e9,
            'current_step_a': i_step,
            'voltage_droop_mv': v_droop * 1000,
            'max_droop_mv': max_droop * 1000,

import numpy as np
import matplotlib.pyplot as plt

class HBMThermalModel:
    """Compact thermal model for HBM stack with RC networks"""
    
    def __init__(self):
        # HBM stack geometry
        self.die_width = 7e-3      # 7 mm
        self.die_height = 5e-3     # 5 mm  
        self.die_thickness = 50e-6  # 50 μm per die
        self.n_dies = 8            # 8 DRAM dies + 1 base die
        self.tsv_diameter = 5e-6   # 5 μm
        self.tsv_pitch = 15e-6     # 15 μm
        
        # Material properties
        self.k_si = 148            # W/m·K (silicon)
        self.k_cu = 400            # W/m·K (copper)
        self.rho_si = 2330         # kg/m³ (silicon)
        self.rho_cu = 8960         # kg/m³ (copper)
        self.cp_si = 700           # J/kg·K (silicon)
        self.cp_cu = 385           # J/kg·K (copper)
        
        # Thermal interface materials
        self.k_tim = 5             # W/m·K (TIM)
        self.tim_thickness = 25e-6 # 25 μm
        
        # Package thermal properties
        self.k_substrate = 20      # W/m·K (package substrate)
        self.k_mold = 1.2          # W/m·K (mold compound)
        
        # Power dissipation (per die)
        self.power_per_die = 2.0   # W (typical HBM3E)
        
    def build_thermal_rc_network(self) -> Dict:
        """Build compact RC thermal network for HBM stack"""
        
        # Thermal resistances (K/W)
        resistances = {}
        capacitances = {}
        
        # Intra-die thermal resistance (lateral)
        die_area = self.die_width * self.die_height
        die_volume = die_area * self.die_thickness
        
        # Lateral resistance within each die
        R_lateral = self.die_thickness / (self.k_si * die_area)
        
        # Vertical resistance through TSVs
        tsv_area = np.pi * (self.tsv_diameter / 2)**2
        n_tsvs = int((self.die_width / self.tsv_pitch) * (self.die_height / self.tsv_pitch))
        total_tsv_area = n_tsvs * tsv_area
        
        R_tsv = self.die_thickness / (self.k_cu * total_tsv_area)
        
        # Parallel combination of silicon and TSV paths
        R_vertical = 1 / (1/R_lateral + 1/R_tsv)
        
        # Thermal capacitances
        C_die = self.rho_si * self.cp_si * die_volume
        
        # Build network for each die
        for i in range(self.n_dies):
            # Lateral resistance within die
            resistances[f'R_lat_{i}'] = R_lateral
            
            # Vertical resistance to next die
            if i < self.n_dies - 1:
                resistances[f'R_vert_{i}_{i+1}'] = R_vertical
            
            # Thermal capacitance
            capacitances[f'C_{i}'] = C_die
        
        # Package thermal resistances
        # Junction to case
        R_jc = 0.5  # K/W (typical for HBM package)
        resistances['R_jc'] = R_jc
        
        # Case to ambient
        R_ca = 15   # K/W (depends on cooling solution)
        resistances['R_ca'] = R_ca
        
        # Package thermal capacitance
        pkg_mass = 0.001  # 1g typical package mass
        pkg_cp = 900      # J/kg·K (average for package materials)
        capacitances['C_pkg'] = pkg_mass * pkg_cp
        
        return {
            'resistances': resistances,
            'capacitances': capacitances,
            'n_nodes': self.n_dies + 2  # Dies + package + ambient
        }
    
    def solve_steady_state_thermal(self, power_profile: List[float]) -> np.ndarray:
        """Solve steady-state thermal distribution"""
        
        if len(power_profile) != self.n_dies:
            raise ValueError(f"Power profile must have {self.n_dies} values")
        
        rc_network = self.build_thermal_rc_network()
        resistances = rc_network['resistances']
        
        # Build conductance matrix
        n_nodes = self.n_dies + 1  # Dies + ambient (reference)
        G = np.zeros((n_nodes, n_nodes))
        I = np.zeros(n_nodes)  # Current (power) sources
        
        # Add power sources
        for i in range(self.n_dies):
            I[i] = power_profile[i]
        
        # Add conductances
        # Lateral conductances (simplified - each die to ambient)
        for i in range(self.n_dies):
            G_total = 1 / (resistances['R_jc'] + resistances['R_ca'])
            G[i, i] += G_total
            G[i, -1] -= G_total  # To ambient node
            G[-1, i] -= G_total
            G[-1, -1] += G_total
        
        # Vertical conductances (die to die)
        for i in range(self.n_dies - 1):
            R_key = f'R_vert_{i}_{i+1}'
            if R_key in resistances:
                G_vert = 1 / resistances[R_key]
                G[i, i] += G_vert
                G[i, i+1] -= G_vert
                G[i+1, i] -= G_vert
                G[i+1, i+1] += G_vert
        
        # Set ambient temperature as reference (0K rise)
        G[-1, -1] = 1e12  # Large number to fix ambient node
        I[-1] = 0
        
        # Solve for temperature rise
        T_rise = spsolve(G, I)
        
        # Add ambient temperature
        T_ambient = 25  # °C
        temperatures = T_rise[:-1] + T_ambient  # Exclude ambient node
        
        return temperatures
    
    def solve_transient_thermal(self, power_profile: List[float], 
                               time_vector: np.ndarray) -> np.ndarray:
        """Solve transient thermal response"""
        
        rc_network = self.build_thermal_rc_network()
        resistances = rc_network['resistances']
        capacitances = rc_network['capacitances']
        
        # Build state-space model: C * dT/dt = -G * T + P
        n_dies = self.n_dies
        
        # Thermal capacitance matrix
        C_matrix = np.zeros((n_dies, n_dies))
        for i in range(n_dies):
            C_matrix[i, i] = capacitances[f'C_{i}']
        
        # Thermal conductance matrix (simplified)
        G_matrix = np.zeros((n_dies, n_dies))
        
        # Thermal coupling between adjacent dies
        for i in range(n_dies - 1):
            R_key = f'R_vert_{i}_{i+1}'
            if R_key in resistances:
                G_coupling = 1 / resistances[R_key]
                G_matrix[i, i] += G_coupling
                G_matrix[i, i+1] -= G_coupling
                G_matrix[i+1, i] -= G_coupling
                G_matrix[i+1, i+1] += G_coupling
        
        # Heat loss to ambient
        R_ambient = resistances['R_jc'] + resistances['R_ca']
        for i in range(n_dies):
            G_matrix[i, i] += 1 / R_ambient
        
        # State-space matrices
        A = -np.linalg.inv(C_matrix) @ G_matrix
        B = np.linalg.inv(C_matrix)
        
        # ODE system
        def thermal_ode(T, t):
            P = np.array(power_profile)  # Constant power for now
            dTdt = A @ T + B @ P
            return dTdt
        
        # Initial conditions (ambient temperature)
        T0 = np.ones(n_dies) * 25  # °C
        
        # Solve ODE
        T_transient = odeint(thermal_ode, T0, time_vector)
        
        return T_transient
    
    def calculate_thermal_coupling_to_performance(self, temperatures: np.ndarray) -> Dict:
        """Calculate impact of temperature on electrical performance"""
        
        # Temperature coefficients for HBM performance
        # Based on typical DRAM characteristics
        temp_coeff_delay = 0.003    # 0.3% per °C
        temp_coeff_jitter = 0.002   # 0.2% per °C  
        temp_coeff_power = 0.008    # 0.8% per °C
        
        T_ref = 25  # °C reference temperature
        
        performance_impact = {}
        
        for i, temp in enumerate(temperatures):
            delta_T = temp - T_ref
            
            # Timing impact
            delay_factor = 1 + temp_coeff_delay * delta_T
            jitter_factor = 1 + temp_coeff_jitter * delta_T
            power_factor = 1 + temp_coeff_power * delta_T
            
            performance_impact[f'die_{i}'] = {
                'temperature_c': temp,
                'delay_factor': delay_factor,
                'jitter_factor': jitter_factor, 
                'power_factor': power_factor,
                'eye_height_degradation': jitter_factor * 0.1,  # Simplified model
                'eye_width_degradation': delay_factor * 0.05
            }
        
        return performance_impact
    
    def design_thermal_management(self, max_temp: float = 85) -> Dict:
        """Design thermal management solution"""
        
        # Start with uniform power
        power_uniform = [self.power_per_die] * self.n_dies
        
        # Calculate temperatures
        temps_steady = self.solve_steady_state_thermal(power_uniform)
        max_temp_achieved = np.max(temps_steady)
        
        recommendations = []
        
        if max_temp_achieved > max_temp:
            # Calculate required thermal resistance reduction
            temp_excess = max_temp_achieved - max_temp
            power_total = sum(power_uniform)
            R_reduction_needed = temp_excess / power_total
            
            recommendations.append(f"Reduce thermal resistance by {R_reduction_needed:.3f} K/W")
            
            # Specific recommendations
            if R_reduction_needed > 5:
                recommendations.append("Consider liquid cooling solution")
            elif R_reduction_needed > 2:
                recommendations.append("Upgrade to high-performance heat sink")
            else:
                recommendations.append("Improve thermal interface material")
        
        # Power throttling analysis
        throttling_temps = []
        throttling_powers = []
        
        for throttle_factor in np.linspace(0.5, 1.0, 6):
            power_throttled = [p * throttle_factor for p in power_uniform]
            temps = self.solve_steady_state_thermal(power_throttled)
            throttling_temps.append(np.max(temps))
            throttling_powers.append(sum(power_throttled))
        
        return {
            'max_temp_no_throttling': max_temp_achieved,
            'thermal_margin': max_temp - max_temp_achieved,
            'recommendations': recommendations,
            'throttling_curve': {
                'temperatures': throttling_temps,
                'powers': throttling_powers
            },
            'die_temperatures': temps_steady.tolist()
        }

Firmware Thermal Governor


import numpy as np
import matplotlib.pyplot as plt
from dataclasses import dataclass
from typing import Dict, List, Tuple
from enum import Enum
import time

class ThermalState(Enum):
    NORMAL = "normal"
    WARNING = "warning" 
    CRITICAL = "critical"
    EMERGENCY = "emergency"

@dataclass
class MemoryConfig:
    """Memory configuration parameters"""
    data_rate: float = 12.8e9      # bps
    burst_length: int = 16
    banks_per_die: int = 16
    dies_per_stack: int = 8
    refresh_rate: float = 64e-3    # 64ms refresh cycle
    
    # Thermal limits
    temp_warning: float = 75       # °C
    temp_critical: float = 85      # °C
    temp_emergency: float = 95     # °C

@dataclass
class PerformanceMetrics:
    """Current performance metrics"""
    bandwidth_gbps: float = 0
    latency_ns: float = 0
    power_watts: float = 0
    temperature_c: float = 0
    throttle_factor: float = 1.0

class HBMThermalGovernor:
    """Firmware thermal governor for HBM memory controller"""
    
    def __init__(self, config: MemoryConfig):
        self.config = config
        self.current_state = ThermalState.NORMAL
        self.thermal_history = []
        self.performance_history = []
        
        # Control parameters
        self.pid_kp = 0.5          # Proportional gain
        self.pid_ki = 0.1          # Integral gain  
        self.pid_kd = 0.05         # Derivative gain
        
        self.integral_error = 0
        self.previous_error = 0
        
        # Throttling parameters
        self.throttle_levels = {
            ThermalState.NORMAL: 1.0,
            ThermalState.WARNING: 0.9,
            ThermalState.CRITICAL: 0.7,
            ThermalState.EMERGENCY: 0.5
        }
        
        # Adaptive parameters
        self.dfe_taps = 8          # DFE tap count
        self.odt_strength = 60     # ODT impedance (ohms)
        self.refresh_rate_multiplier = 1.0
        
    def update_thermal_state(self, temperature: float) -> ThermalState:
        """Update thermal state based on temperature"""
        
        if temperature >= self.config.temp_emergency:
            new_state = ThermalState.EMERGENCY
        elif temperature >= self.config.temp_critical:
            new_state = ThermalState.CRITICAL
        elif temperature >= self.config.temp_warning:
            new_state = ThermalState.WARNING
        else:
            new_state = ThermalState.NORMAL
        
        # Log state transitions
        if new_state != self.current_state:
            print(f"Thermal state transition: {self.current_state.value} -> {new_state.value}")
            print(f"Temperature: {temperature:.1f}°C")
        
        self.current_state = new_state
        return new_state
    
    def calculate_pid_control(self, target_temp: float, current_temp: float, dt: float) -> float:
        """PID controller for thermal management"""
        
        error = target_temp - current_temp
        
        # Proportional term
        p_term = self.pid_kp * error
        
        # Integral term
        self.integral_error += error * dt
        i_term = self.pid_ki * self.integral_error
        
        # Derivative term
        d_error = (error - self.previous_error) / dt
        d_term = self.pid_kd * d_error
        
        # PID output
        pid_output = p_term + i_term + d_term
        
        # Update for next iteration
        self.previous_error = error
        
        # Clamp output
        return np.clip(pid_output, -1.0, 1.0)
    
    def adaptive_refresh_control(self, temperature: float) -> float:
        """Adaptive refresh rate based on temperature"""
        
        # Higher temperature requires faster refresh
        # Arrhenius relationship for retention time
        Ea = 0.7  # eV (activation energy)
        k = 8.617e-5  # eV/K (Boltzmann constant)
        T_ref = 298.15  # K (25°C reference)
        T_current = temperature + 273.15  # Convert to Kelvin
        
        # Retention time scaling
        retention_factor = np.exp(Ea / k * (1/T_current - 1/T_ref))
        
        # Refresh rate is inverse of retention time
        refresh_multiplier = 1 / retention_factor
        
        # Limit refresh rate increase
        refresh_multiplier = np.clip(refresh_multiplier, 1.0, 4.0)
        
        return refresh_multiplier
    
    def optimize_signal_integrity(self, temperature: float, data_rate: float) -> Dict:
        """Optimize SI parameters based on thermal conditions"""
        
        # Temperature effects on signal integrity
        temp_delta = temperature - 25  # °C from reference
        
        # DFE optimization
        # Higher temperature increases jitter, need more DFE taps
        optimal_dfe_taps = self.dfe_taps + int(temp_delta / 10)
        optimal_dfe_taps = np.clip(optimal_dfe_taps, 4, 16)
        
        # ODT optimization
        # Higher temperature increases driver resistance
        temp_coeff_odt = 0.002  # 0.2% per °C
        odt_factor = 1 + temp_coeff_odt * temp_delta
        optimal_odt = self.odt_strength * odt_factor
        optimal_odt = np.clip(optimal_odt, 40, 120)  # Ohms
        
        # Voltage scaling
        # Higher temperature may require voltage boost
        vdd_nominal = 1.2  # V
        temp_coeff_voltage = 0.001  # 0.1% per °C
        voltage_factor = 1 + temp_coeff_voltage * temp_delta
        optimal_vdd = vdd_nominal * voltage_factor
        optimal_vdd = np.clip(optimal_vdd, 1.1, 1.35)  # V
        
        return {
            'dfe_taps': optimal_dfe_taps,
            'odt_impedance': optimal_odt,
            'supply_voltage': optimal_vdd,
            'eye_margin_estimate': self._estimate_eye_margin(temperature, data_rate)
        }
    
    def _estimate_eye_margin(self, temperature: float, data_rate: float) -> Dict:
        """Estimate eye diagram margins"""
        
        # Simplified eye margin model
        temp_delta = temperature - 25
        
        # Base eye margins at 25°C
        eye_height_base = 200  # mV
        eye_width_base = 0.7   # UI
        
        # Temperature degradation
        height_degradation = 0.02 * temp_delta  # 2% per °C
        width_degradation = 0.015 * temp_delta   # 1.5% per °C
        
        eye_height = eye_height_base * (1 - height_degradation / 100)
        eye_width = eye_width_base * (1 - width_degradation / 100)
        
        # Data rate effects
        rate_factor = data_rate / 12.8e9  # Normalized to HBM3E baseline
        eye_width *= (1 / rate_factor)    # Higher rate reduces eye width
        
        return {
            'eye_height_mv': max(eye_height, 50),    # Minimum 50mV
            'eye_width_ui': max(eye_width, 0.3),     # Minimum 0.3 UI
            'timing_margin_ps': max(eye_width * 78, 25)  # Convert to ps
        }
    
    def bandwidth_throttling_decision(self, temperature: float, 
                                    target_bandwidth: float) -> Dict:
        """Make bandwidth throttling decisions"""
        
        # Get base throttle factor from thermal state
        base_throttle = self.throttle_levels[self.current_state]
        
        # PID control for fine adjustment
        target_temp = self.config.temp_warning - 5  # 5°C margin
        pid_adjustment = self.calculate_pid_control(target_temp, temperature, 0.1)
        
        # Combine base throttling with PID adjustment
        final_throttle = base_throttle + pid_adjustment * 0.1
        final_throttle = np.clip(final_throttle, 0.3, 1.0)
        
        # Calculate actual bandwidth
        actual_bandwidth = target_bandwidth * final_throttle
        
        # Refresh rate adjustment
        refresh_multiplier = self.adaptive_refresh_control(temperature)
        
        # Power estimation
        power_base = 16  # W (typical HBM3E)
        power_bandwidth_factor = final_throttle
        power_refresh_factor = refresh_multiplier
        estimated_power = power_base * power_bandwidth_factor * power_refresh_factor
        
        return {
            'throttle_factor': final_throttle,
            'bandwidth_gbps': actual_bandwidth / 1e9,
            'refresh_multiplier': refresh_multiplier,
            'estimated_power_w': estimated_power,
            'thermal_state': self.current_state.value,
            'pid_adjustment': pid_adjustment
        }
    
    def run_thermal_management_loop(self, temperature_profile: List[float],
                                  time_steps: List[float]) -> Dict:
        """Run complete thermal management simulation"""
        
        results = {
            'time': time_steps,
            'temperatures': temperature_profile,
            'throttle_factors': [],
            'bandwidths': [],
            'power_consumption': [],
            'thermal_states': [],
            'eye_margins': []
        }
        
        target_bandwidth = 1600e9  # 1.6 TB/s (HBM3E target)
        
        for i, (temp, t) in enumerate(zip(temperature_profile, time_steps)):
            # Update thermal state
            thermal_state = self.update_thermal_state(temp)
            
            # Make throttling decision
            throttle_decision = self.bandwidth_throttling_decision(temp, target_bandwidth)
            
            # Optimize signal integrity
            si_optimization = self.optimize_signal_integrity(temp, self.config.data_rate)
            
            # Store results
            results['throttle_factors'].append(throttle_decision['throttle_factor'])
            results['bandwidths'].append(throttle_decision['bandwidth_gbps'])
            results['power_consumption'].append(throttle_decision['estimated_power_w'])
            results['thermal_states'].append(throttle_decision['thermal_state'])
            results['eye_margins'].append(si_optimization['eye_margin_estimate'])
            
            # Update thermal history
            self.thermal_history.append({
                'time': t,
                'temperature': temp,
                'state': thermal_state,
                'throttle': throttle_decision['throttle_factor']
            })
        
        return results
    
    def generate_performance_report(self, simulation_results: Dict) -> Dict:
        """Generate comprehensive performance report"""
        
        temps = np.array(simulation_results['temperatures'])
        throttles = np.array(simulation_results['throttle_factors'])
        bandwidths = np.array(simulation_results['bandwidths'])
        powers = np.array(simulation_results['power_consumption'])
        
        # Calculate metrics
        avg_throttle = np.mean(throttles)
        min_throttle = np.min(throttles)
        throttle_events = np.sum(throttles < 0.95)  # Count significant throttling
        
        avg_bandwidth = np.mean(bandwidths)
        bandwidth_efficiency = avg_bandwidth / 1600  # vs target 1.6 TB/s
        
        avg_power = np.mean(powers)
        peak_power = np.max(powers)
        
        max_temp = np.max(temps)
        avg_temp = np.mean(temps)
        
        # Performance/Watt calculation
        perf_per_watt = avg_bandwidth / avg_power
        
        report = {
            'thermal_performance': {
                'max_temperature_c': max_temp,
                'avg_temperature_c': avg_temp,
                'thermal_violations': np.sum(temps > 85)
            },
            'bandwidth_performance': {
                'avg_bandwidth_gbps': avg_bandwidth,
                'bandwidth_efficiency_percent': bandwidth_efficiency * 100,
                'throttle_events': int(throttle_events),
                'avg_throttle_factor': avg_throttle,
                'min_throttle_factor': min_throttle
            },
            'power_performance': {
                'avg_power_w': avg_power,
                'peak_power_w': peak_power,
                'performance_per_watt': perf_per_watt
            },
            'overall_score': bandwidth_efficiency * (1 - throttle_events / len(temps))
        }
        
        return report
