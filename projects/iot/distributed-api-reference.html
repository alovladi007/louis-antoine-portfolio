Distributed Trainer

DistributedTrainer(config)

Main trainer class for distributed training with automatic parallelism.

from distributed_ml import DistributedTrainer, TrainingConfig

config = TrainingConfig(
    model_name="llama-7b",
    batch_size=32,
    gradient_accumulation_steps=4,
    learning_rate=1e-4,
    max_steps=10000,

    # Parallelism settings
    tensor_parallel_size=4,
    pipeline_parallel_size=2,
    zero_stage=2,

    # Optimizations
    mixed_precision="bf16",
    activation_checkpointing=True,
    gradient_bucketing=True
)

trainer = DistributedTrainer(config)
trainer.train(dataset)
Key Methods:
  • train(dataset) - Start distributed training
  • evaluate(dataset) - Run distributed evaluation
  • save_checkpoint(path) - Save distributed checkpoint
  • load_checkpoint(path) - Load from checkpoint

Process Group Management

init_distributed()

Initialize distributed training environment.

from distributed_ml import init_distributed, get_world_size, get_rank

# Auto-detect backend and initialize
init_distributed(
    backend="nccl",  # or "gloo" for CPU
    init_method="env://",  # or "tcp://master:port"
    world_size=None,  # auto-detect from env
    rank=None  # auto-detect from env
)

# Query process group info
print(f"World size: {get_world_size()}")
print(f"Global rank: {get_rank()}")
print(f"Local rank: {get_local_rank()}")
print(f"Is main process: {is_main_process()}")

create_process_groups()

Create hierarchical process groups for 3D parallelism.

from distributed_ml import create_process_groups

groups = create_process_groups(
    tensor_parallel_size=4,
    pipeline_parallel_size=2,
    data_parallel_size=4  # auto-computed if None
)

# Access specific groups
tp_group = groups.tensor_parallel_group
pp_group = groups.pipeline_parallel_group
dp_group = groups.data_parallel_group

# Per-group operations
tp_all_reduce(tensor, group=tp_group)

Model Parallelism

TensorParallelLinear

Tensor-parallel linear layer with column or row splitting.

from distributed_ml.layers import (
    TensorParallelLinear,
    ColumnParallelLinear,
    RowParallelLinear
)

# Column parallel: split output features
col_linear = ColumnParallelLinear(
    in_features=4096,
    out_features=16384,
    bias=True,
    gather_output=False  # keep distributed
)

# Row parallel: split input features
row_linear = RowParallelLinear(
    in_features=16384,
    out_features=4096,
    bias=True,
    input_is_parallel=True
)

# MLP pattern: col -> activation -> row
hidden = F.gelu(col_linear(x))
output = row_linear(hidden)  # all-reduce happens here

PipelineModule

Wrap model for pipeline parallelism with automatic layer splitting.

from distributed_ml import PipelineModule, PipelineSchedule

# Define layer partition
model = PipelineModule(
    layers=[
        *[TransformerBlock(i) for i in range(32)]
    ],
    num_stages=4,
    partition_method="uniform"  # or "parameters", "memory"
)

# Configure schedule
schedule = PipelineSchedule(
    micro_batches=8,
    schedule="1f1b"  # or "gpipe", "interleaved"
)

# Execute pipeline
output = model(input, schedule=schedule)

ZeRO Optimizer

ZeROOptimizer

Memory-efficient optimizer with state partitioning.

from distributed_ml import ZeROOptimizer

optimizer = ZeROOptimizer(
    params=model.parameters(),
    optimizer_class=torch.optim.AdamW,

    # ZeRO configuration
    stage=3,  # 1, 2, or 3
    reduce_bucket_size=5e8,
    allgather_bucket_size=5e8,

    # CPU offload (ZeRO-Infinity)
    offload_optimizer=True,
    offload_params=True,

    # Optimizer kwargs
    lr=1e-4,
    weight_decay=0.01,
    betas=(0.9, 0.95)
)

# Training loop
for batch in dataloader:
    loss = model(batch)
    loss.backward()
    optimizer.step()
    optimizer.zero_grad()

FSDP Wrapper

PyTorch native FSDP with automatic sharding.

from distributed_ml import wrap_fsdp, FSDPConfig

config = FSDPConfig(
    sharding_strategy="FULL_SHARD",  # SHARD_GRAD_OP, NO_SHARD
    mixed_precision="bf16",
    activation_checkpointing=True,

    # Auto-wrap policy
    auto_wrap_policy="transformer_layer",
    transformer_layer_cls=TransformerBlock,
    min_num_params=1e6,

    # CPU offload
    cpu_offload=False
)

model = wrap_fsdp(model, config)

Communication Primitives

Collective Operations

from distributed_ml.comm import (
    all_reduce,
    all_gather,
    reduce_scatter,
    broadcast
)

# All-reduce: sum gradients across all ranks
all_reduce(tensor, op="sum", group=None)

# All-gather: gather tensors from all ranks
gathered = all_gather(tensor, group=dp_group)  # [world_size, ...]

# Reduce-scatter: reduce and scatter chunks
output = reduce_scatter(tensor, op="sum", group=dp_group)

# Broadcast from rank 0
broadcast(tensor, src=0, group=None)

# Async operations
handle = all_reduce(tensor, async_op=True)
# ... do other work ...
handle.wait()

Checkpointing

DistributedCheckpoint

from distributed_ml import DistributedCheckpoint

checkpoint = DistributedCheckpoint(
    save_dir="checkpoints/",
    save_interval=1000,  # steps
    keep_last_n=3,

    # Sharded saving for large models
    sharded=True,
    async_save=True
)

# Save checkpoint
checkpoint.save(
    model=model,
    optimizer=optimizer,
    step=global_step,
    metrics={"loss": loss.item()}
)

# Load checkpoint
state = checkpoint.load(
    path="checkpoints/step_5000",
    model=model,
    optimizer=optimizer,
    strict=True
)
start_step = state["step"]

Data Loading

DistributedDataLoader

from distributed_ml import DistributedDataLoader

dataloader = DistributedDataLoader(
    dataset=dataset,
    batch_size=32,

    # Distributed sampling
    shuffle=True,
    drop_last=True,

    # Preprocessing
    num_workers=4,
    pin_memory=True,
    prefetch_factor=2,

    # Sequence packing (for LLMs)
    pack_sequences=True,
    max_seq_length=2048
)

# Resume from step
dataloader.set_epoch(epoch)
dataloader.skip_batches(start_step)

Profiling & Monitoring

DistributedProfiler

from distributed_ml import DistributedProfiler

profiler = DistributedProfiler(
    enabled=True,
    output_dir="profiles/",

    # What to profile
    profile_cuda=True,
    profile_memory=True,
    profile_comm=True,

    # Timing
    warmup_steps=10,
    active_steps=5,
    repeat=1
)

with profiler:
    for step, batch in enumerate(dataloader):
        loss = model(batch)
        loss.backward()
        optimizer.step()

# Export results
profiler.export_chrome_trace("trace.json")
profiler.export_tensorboard("tb_logs/")
print(profiler.summary())
