Quick Start

from genscientific import MoleculeVAE, MoleculeDataset

# Load pre-trained model
model = MoleculeVAE.from_pretrained("zinc250k-vae")

# Generate new molecules
molecules = model.sample(n_samples=100)

# Conditional generation
molecules = model.sample(
    n_samples=50,
    conditions={"logp": 2.5, "mw": 350}
)

Models

MoleculeVAE

Variational Autoencoder for molecular generation with SMILES representation.

class MoleculeVAE(nn.Module):
    """
    Variational Autoencoder for molecular generation.

    Args:
        vocab_size (int): Size of SMILES vocabulary
        latent_dim (int): Dimension of latent space (default: 256)
        hidden_dim (int): Hidden layer dimension (default: 512)
        num_layers (int): Number of encoder/decoder layers (default: 3)
        dropout (float): Dropout probability (default: 0.1)
    """

    def __init__(
        self,
        vocab_size: int,
        latent_dim: int = 256,
        hidden_dim: int = 512,
        num_layers: int = 3,
        dropout: float = 0.1
    ): ...

    def encode(self, x: Tensor) -> Tuple[Tensor, Tensor]:
        """Encode input to latent distribution parameters."""

    def decode(self, z: Tensor) -> Tensor:
        """Decode latent vector to SMILES logits."""

    def sample(
        self,
        n_samples: int,
        temperature: float = 1.0,
        conditions: Optional[Dict] = None
    ) -> List[str]:
        """Generate new molecules."""

    def reconstruct(self, smiles: List[str]) -> List[str]:
        """Reconstruct molecules from SMILES."""

MoleculeDiffusion

Denoising Diffusion model for 3D molecular generation with E(3) equivariance.

class MoleculeDiffusion(nn.Module):
    """
    E(3) Equivariant Diffusion Model for 3D molecules.

    Args:
        num_atom_types (int): Number of atom types
        hidden_dim (int): Hidden dimension (default: 256)
        num_layers (int): Number of EGNN layers (default: 6)
        num_timesteps (int): Diffusion timesteps (default: 1000)
    """

    def sample(
        self,
        n_samples: int,
        n_atoms: int,
        guidance_scale: float = 1.0
    ) -> Tuple[Tensor, Tensor]:
        """Generate 3D molecular structures."""

    def log_prob(
        self,
        positions: Tensor,
        atom_types: Tensor
    ) -> Tensor:
        """Compute log probability of molecules."""

MoleculeFlow

Normalizing Flow model with continuous normalizing flows for molecules.

class MoleculeFlow(nn.Module):
    """
    Continuous Normalizing Flow for molecular generation.

    Args:
        input_dim (int): Input feature dimension
        hidden_dim (int): Hidden dimension (default: 256)
        num_blocks (int): Number of flow blocks (default: 4)
    """

    def forward(self, x: Tensor) -> Tuple[Tensor, Tensor]:
        """Forward pass returning transformed x and log_det."""

    def inverse(self, z: Tensor) -> Tensor:
        """Inverse transform from latent to data space."""

    def sample(self, n_samples: int) -> Tensor:
        """Sample from the learned distribution."""

Training

from genscientific import Trainer, TrainingConfig

config = TrainingConfig(
    model_type="vae",
    latent_dim=256,
    batch_size=128,
    learning_rate=1e-4,
    num_epochs=100,
    kl_weight=0.1,
    gradient_clip=1.0,
    checkpoint_dir="./checkpoints"
)

trainer = Trainer(model, dataset, config)
trainer.train()

# Resume training
trainer.resume("./checkpoints/epoch_50.pt")

Evaluation

from genscientific.evaluation import (
    compute_validity,
    compute_uniqueness,
    compute_novelty,
    compute_fcd,
    evaluate_all
)

# Individual metrics
validity = compute_validity(generated_smiles)
uniqueness = compute_uniqueness(generated_smiles)
novelty = compute_novelty(generated_smiles, training_smiles)
fcd = compute_fcd(generated_smiles, reference_smiles)

# All metrics at once
results = evaluate_all(
    generated_smiles,
    reference_smiles=training_smiles,
    metrics=["validity", "uniqueness", "novelty", "fcd", "kl_div"]
)
print(results)
