Drug Discovery

Pharmaceutical Applications

Generate novel drug candidates targeting specific proteins with optimized ADMET properties.

# Target-aware molecule generation
from genscientific import DrugGenerator

generator = DrugGenerator.from_pretrained("drug-vae-v2")

candidates = generator.generate(
    target_protein="ACE2",
    n_samples=1000,
    constraints={
        "mw": (300, 500),
        "logp": (1, 4),
        "hbd": (0, 5),
        "pains_filter": True
    }
)

# Rank by predicted binding affinity
ranked = generator.rank_by_binding(candidates, target="ACE2")
print(f"Top candidate: {ranked[0].smiles}")

Battery Materials

Materials Science

Design new solid-state electrolyte materials with high ionic conductivity.

# Solid electrolyte generation
from genscientific import CrystalDiffusion

model = CrystalDiffusion.from_pretrained("mat-diffusion-v1")

crystals = model.generate(
    n_samples=500,
    composition_constraint=["Li", "P", "S", "Cl"],
    property_targets={
        "ionic_conductivity": (1e-3, 1e-1),  # S/cm
        "electrochemical_stability": (0, 5)   # V
    }
)

# Validate with DFT
for crystal in crystals[:10]:
    stability = model.predict_stability(crystal)
    print(f"{crystal.formula}: {stability:.3f} eV/atom")

Protein Engineering

Biotechnology

Design enzyme variants with enhanced stability and catalytic activity.

# Enzyme optimization
from genscientific import ProteinVAE

model = ProteinVAE.from_pretrained("esm-vae")

# Start from wild-type enzyme
wt_sequence = "MVLSPADKTN..."

variants = model.optimize(
    seed_sequence=wt_sequence,
    objectives={
        "thermal_stability": "maximize",
        "activity": "maximize",
        "expression": "maximize"
    },
    n_generations=20,
    mutations_per_variant=5
)

# Analyze top variants
for v in variants[:5]:
    print(f"Mutations: {v.mutations}")
    print(f"Predicted Tm: {v.predicted_tm:.1f}°C")

Catalyst Design

Chemical Engineering

Generate novel catalysts for CO2 reduction and green chemistry applications.

# Heterogeneous catalyst generation
from genscientific import SurfaceGenerator

model = SurfaceGenerator.from_pretrained("cat-gen-v1")

catalysts = model.generate(
    reaction="CO2 + H2 -> CH3OH + H2O",
    base_materials=["Cu", "Zn", "Al"],
    n_samples=200,
    optimize_for=["selectivity", "activity"]
)

# Predict adsorption energies
for cat in catalysts[:5]:
    ads_E = model.predict_adsorption(cat, adsorbate="CO2")
    print(f"{cat.composition}: ΔE_ads = {ads_E:.2f} eV")

MOF Discovery

Porous Materials

Design Metal-Organic Frameworks for gas storage and separation applications.

# MOF generation for hydrogen storage
from genscientific import MOFGenerator

model = MOFGenerator.from_pretrained("mof-diffusion")

mofs = model.generate(
    n_samples=100,
    targets={
        "surface_area": (2000, 5000),  # m²/g
        "pore_volume": (0.5, 2.0),      # cm³/g
        "h2_uptake_77K": "maximize"
    },
    metal_nodes=["Zn", "Cu", "Zr"],
    linkers=["BDC", "BTC", "NDC"]
)

# Simulate gas uptake
for mof in mofs[:3]:
    uptake = model.simulate_adsorption(mof, gas="H2", T=77)
    print(f"{mof.name}: {uptake:.1f} wt%")

Polymer Design

Materials Science

Generate novel polymers with targeted mechanical and thermal properties.

# High-performance polymer generation
from genscientific import PolymerVAE

model = PolymerVAE.from_pretrained("poly-vae-v2")

polymers = model.generate(
    n_samples=500,
    property_targets={
        "Tg": (150, 250),        # Glass transition (°C)
        "tensile_strength": (50, 100),  # MPa
        "thermal_conductivity": (0.1, 0.5)  # W/m·K
    },
    monomer_constraints={
        "max_atoms": 30,
        "element_whitelist": ["C", "H", "O", "N"]
    }
)

# Predict properties
for p in polymers[:5]:
    props = model.predict_properties(p)
    print(f"{p.smiles}: Tg={props['Tg']:.0f}°C")
