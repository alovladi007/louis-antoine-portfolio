API Reference

Python API documentation for neural signal processing

class SpikeSorter

Core Module

Automatic spike detection, feature extraction, and clustering for unit isolation.

from neuraldecode import SpikeSorter

sorter = SpikeSorter(sample_rate=30000)
sorter.load_data(raw_signal)  # (channels, samples)

# Detect and sort spikes
sorter.detect_spikes(threshold=-4.0)
sorter.extract_features(method='pca', n_components=3)
sorter.cluster(method='gmm', n_clusters=4)

# Get results
units = sorter.get_units()
print(f"Found {len(units)} units")
detect_spikes(threshold=-4.0, method="threshold")

Detect spikes using threshold crossing or template matching.

extract_features(method="pca", n_components=3)

Extract PCA, wavelet, or waveform features from detected spikes.

cluster(method="gmm", n_clusters=None)

Cluster spikes into units using GMM, K-means, or DBSCAN.

get_quality_metrics()

Returns isolation distance, L-ratio, and ISI violations per unit.

class NeuralDecoder

Decoding

Population-level decoders for motor intent estimation.

from neuraldecode import NeuralDecoder

decoder = NeuralDecoder(n_units=64, output_dims=3)

# Train on calibration data
decoder.fit(spike_rates, kinematics, method='kalman')

# Real-time decoding
velocity = decoder.decode(current_rates)
print(f"Decoded velocity: {velocity}")
fit(spike_rates, kinematics, method="kalman")

Train decoder on calibration data. Methods: kalman, wiener, pva, lstm.

decode(spike_rates)

Decode single time bin of spike rates to kinematic output.

update_online(spike_rates, intended_kinematics)

Perform online adaptation using intention estimation (ReFIT).

class TuningAnalyzer

Analysis 
from neuraldecode import TuningAnalyzer

analyzer = TuningAnalyzer()

# Fit tuning curve
tuning = analyzer.fit_tuning(
    firing_rates,  # (n_directions,)
    directions,    # angles in radians
    model='cosine'
)

print(f"Preferred direction: {tuning['pd']:.1f}°")
print(f"Modulation depth: {tuning['depth']:.1f} Hz")
fit_tuning(rates, directions, model="cosine")

Fit cosine, von Mises, or Gaussian tuning model.

compute_population_vector(rates, preferred_dirs)

Calculate population vector from ensemble activity.

class RealtimeProcessor

Real-Time 
from neuraldecode import RealtimeProcessor

processor = RealtimeProcessor(
    sample_rate=30000,
    n_channels=96,
    bin_size_ms=50
)

# Configure pipeline
processor.set_filter(300, 6000)  # bandpass
processor.set_threshold(-4.0)
processor.set_decoder(trained_decoder)

# Process streaming data
for chunk in data_stream:
    result = processor.process(chunk)
    if result.has_output:
        send_to_device(result.velocity)
process(data_chunk)

Process streaming data through full pipeline. Returns decoded output when bin completes.

get_timing_stats()

Returns processing latency statistics per pipeline stage.

class ElectrodeArray

Hardware 
from neuraldecode import ElectrodeArray

array = ElectrodeArray.utah_96()  # 10x10, 96 channels
# or
array = ElectrodeArray.neuropixels_1()  # 384 channels

# Get channel positions
positions = array.get_positions()  # (n_channels, 2) in µm

# Check impedances
array.set_impedances(measured_values)
bad_channels = array.get_bad_channels(threshold=1000)  # kΩ
utah_96() / utah_64() / neuropixels_1()

Factory methods for common electrode array configurations.

get_channel_neighbors(channel_id, radius)

Find spatially adjacent channels for local referencing.
