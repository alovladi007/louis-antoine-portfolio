"""
Complete Spike Sorting Pipeline
Detection → Feature Extraction → Clustering → Quality Assessment
"""

import numpy as np
from scipy.signal import butter, filtfilt
from sklearn.decomposition import PCA
from sklearn.mixture import GaussianMixture
from scipy.spatial.distance import mahalanobis


class SpikeSorter:
    """Production spike sorting with quality metrics"""

    def __init__(self, fs=30000):
        self.fs = fs
        self.threshold_factor = 4.0
        self.pre_samples = 10
        self.post_samples = 30

    def filter(self, data, low=300, high=6000):
        """Bandpass filter for spike band"""
        nyq = self.fs / 2
        b, a = butter(4, [low/nyq, high/nyq], btype='band')
        return filtfilt(b, a, data)

    def estimate_noise(self, data):
        """Robust noise estimation using MAD"""
        return np.median(np.abs(data)) / 0.6745

    def detect_spikes(self, data):
        """Threshold-based spike detection"""
        noise_std = self.estimate_noise(data)
        threshold = -self.threshold_factor * noise_std

        # Find threshold crossings
        crossings = np.where(data < threshold)[0]

        # Apply dead time (1ms)
        dead_samples = int(0.001 * self.fs)
        spike_times = []
        last = -dead_samples

        for idx in crossings:
            if idx - last >= dead_samples:
                # Find local minimum
                start = max(0, idx - 5)
                end = min(len(data), idx + 5)
                local_min = start + np.argmin(data[start:end])
                spike_times.append(local_min)
                last = local_min

        return np.array(spike_times)

    def extract_waveforms(self, data, spike_times):
        """Extract aligned spike waveforms"""
        waveforms = []
        valid_times = []

        for t in spike_times:
            if t - self.pre_samples >= 0 and t + self.post_samples < len(data):
                wf = data[t - self.pre_samples:t + self.post_samples]
                waveforms.append(wf)
                valid_times.append(t)

        return np.array(waveforms), np.array(valid_times)

    def cluster(self, waveforms, n_components=3, max_clusters=8):
        """PCA + GMM clustering with BIC selection"""
        # PCA
        pca = PCA(n_components=n_components)
        features = pca.fit_transform(waveforms)

        # Find optimal cluster count via BIC
        best_bic = np.inf
        best_gmm = None

        for n in range(1, max_clusters + 1):
            gmm = GaussianMixture(n_components=n, random_state=42)
            gmm.fit(features)
            bic = gmm.bic(features)
            if bic < best_bic:
                best_bic = bic
                best_gmm = gmm

        labels = best_gmm.predict(features)

        return labels, features, pca, best_gmm

    def compute_quality_metrics(self, features, labels, gmm):
        """Compute isolation distance and L-ratio"""
        metrics = {}
        n_clusters = len(np.unique(labels))

        for cluster_id in range(n_clusters):
            mask = labels == cluster_id
            cluster_features = features[mask]
            other_features = features[~mask]

            if len(cluster_features) < 2 or len(other_features) < 1:
                continue

            # Cluster statistics
            mean = np.mean(cluster_features, axis=0)
            cov = np.cov(cluster_features.T)
            cov_inv = np.linalg.pinv(cov)

            # Isolation distance
            distances = [mahalanobis(f, mean, cov_inv) for f in other_features]
            iso_dist = np.percentile(distances, 50) if len(distances) > 0 else 0

            metrics[cluster_id] = {
                'isolation_distance': iso_dist,
                'n_spikes': int(np.sum(mask)),
                'mean_waveform': mean
            }

        return metrics

    def sort(self, data):
        """Complete sorting pipeline"""
        # Filter
        filtered = self.filter(data)

        # Detect
        spike_times = self.detect_spikes(filtered)

        # Extract waveforms
        waveforms, valid_times = self.extract_waveforms(filtered, spike_times)

        # Cluster
        labels, features, pca, gmm = self.cluster(waveforms)

        # Quality metrics
        metrics = self.compute_quality_metrics(features, labels, gmm)

        return {
            'spike_times': valid_times,
            'labels': labels,
            'waveforms': waveforms,
            'features': features,
            'metrics': metrics,
            'n_units': len(np.unique(labels))
        }

"""
Kalman Filter Decoder for Motor BCI
Decodes 2D/3D velocity from neural population activity
"""

import numpy as np


class KalmanVelocityDecoder:
    """Kalman filter for velocity decoding"""

    def __init__(self, n_units, state_dim=6):
        """
        state_dim: [x, y, z, vx, vy, vz] (position + velocity)
        """
        self.n_units = n_units
        self.state_dim = state_dim

        # State transition (constant velocity model)
        self.A = np.eye(state_dim)
        self.A[:3, 3:] = np.eye(3) * 0.05  # dt = 50ms

        # Observation model (fit during training)
        self.H = np.zeros((n_units, state_dim))

        # Noise covariances
        self.W = np.eye(state_dim) * 0.001  # Process noise
        self.Q = np.eye(n_units)  # Observation noise

        # State estimate
        self.x = np.zeros(state_dim)
        self.P = np.eye(state_dim)

    def fit(self, spike_rates, kinematics):
        """Fit decoder from calibration data"""
        # spike_rates: (T, n_units)
        # kinematics: (T, state_dim)

        Y = spike_rates
        X = kinematics

        # Fit H via least squares: Y = H @ X
        self.H = np.linalg.lstsq(X, Y, rcond=None)[0].T

        # Estimate observation noise
        residuals = Y - X @ self.H.T
        self.Q = np.cov(residuals.T) + np.eye(self.n_units) * 1e-6

        # Fit process noise from state transitions
        state_residuals = X[1:] - X[:-1] @ self.A.T
        self.W = np.cov(state_residuals.T) + np.eye(self.state_dim) * 1e-6

    def reset(self):
        """Reset state estimate"""
        self.x = np.zeros(self.state_dim)
        self.P = np.eye(self.state_dim)

    def decode(self, y):
        """Single-step Kalman update"""
        # Predict
        x_pred = self.A @ self.x
        P_pred = self.A @ self.P @ self.A.T + self.W

        # Kalman gain
        S = self.H @ P_pred @ self.H.T + self.Q
        K = P_pred @ self.H.T @ np.linalg.inv(S)

        # Update
        innovation = y - self.H @ x_pred
        self.x = x_pred + K @ innovation
        self.P = (np.eye(self.state_dim) - K @ self.H) @ P_pred

        return self.x[3:6]  # Return velocity only

"""
Real-Time BCI Control System
Closed-loop cursor/prosthetic control with minimal latency
"""

import numpy as np
from scipy.signal import lfilter, lfilter_zi, butter
import time


class RealtimeBCI:
    """Low-latency BCI control system"""

    def __init__(self, decoder, n_channels=96, fs=30000, bin_ms=50):
        self.decoder = decoder
        self.n_channels = n_channels
        self.fs = fs
        self.bin_samples = int(bin_ms * fs / 1000)

        # Causal bandpass filter
        b, a = butter(4, [300/15000, 6000/15000], btype='band')
        self.b, self.a = b, a
        self.filter_states = [lfilter_zi(b, a) for _ in range(n_channels)]

        # Spike detection thresholds (per channel)
        self.thresholds = np.zeros(n_channels)

        # Spike count buffer
        self.spike_counts = np.zeros(n_channels)
        self.sample_count = 0

        # Timing stats
        self.timing_log = []

    def calibrate_thresholds(self, calibration_data, factor=4.0):
        """Set detection thresholds from noise estimate"""
        for ch in range(self.n_channels):
            noise_std = np.median(np.abs(calibration_data[ch])) / 0.6745
            self.thresholds[ch] = -factor * noise_std

    def process_chunk(self, chunk):
        """Process chunk of samples, return velocity when bin complete"""
        t_start = time.perf_counter()

        for sample_idx in range(chunk.shape[1]):
            for ch in range(self.n_channels):
                # Causal filter
                filtered, self.filter_states[ch] = lfilter(
                    self.b, self.a, [chunk[ch, sample_idx]],
                    zi=self.filter_states[ch]
                )

                # Threshold detection
                if filtered[0] < self.thresholds[ch]:
                    self.spike_counts[ch] += 1

            self.sample_count += 1

            # Check if bin complete
            if self.sample_count >= self.bin_samples:
                # Convert to rates (Hz)
                bin_duration = self.bin_samples / self.fs
                rates = self.spike_counts / bin_duration

                # Decode
                velocity = self.decoder.decode(rates)

                # Reset for next bin
                self.spike_counts[:] = 0
                self.sample_count = 0

                # Log timing
                elapsed = (time.perf_counter() - t_start) * 1000
                self.timing_log.append(elapsed)

                return velocity

        return None

    def get_timing_stats(self):
        """Return processing timing statistics"""
        if len(self.timing_log) == 0:
            return {}
        return {
            'mean_ms': np.mean(self.timing_log),
            'max_ms': np.max(self.timing_log),
            'std_ms': np.std(self.timing_log)
        }

def fit_cosine_tuning(firing_rates, directions):
    """
    Fit cosine tuning: f(θ) = b + m*cos(θ - θ_pd)

    Parameters:
        firing_rates: array of mean rates per direction
        directions: array of movement directions (radians)

    Returns:
        dict with baseline, modulation, preferred_direction, r_squared
    """
    # Build design matrix: [1, cos(θ), sin(θ)]
    X = np.column_stack([
        np.ones(len(directions)),
        np.cos(directions),
        np.sin(directions)
    ])

    # Least squares fit
    coeffs, residuals, _, _ = np.linalg.lstsq(X, firing_rates, rcond=None)
    b, c1, c2 = coeffs

    # Extract parameters
    baseline = b
    modulation = np.sqrt(c1**2 + c2**2)
    preferred_direction = np.arctan2(c2, c1)

    # R-squared
    ss_res = np.sum((firing_rates - X @ coeffs)**2)
    ss_tot = np.sum((firing_rates - np.mean(firing_rates))**2)
    r_squared = 1 - ss_res / ss_tot if ss_tot > 0 else 0

    return {
        'baseline': baseline,
        'modulation': modulation,
        'preferred_direction': np.degrees(preferred_direction),
        'r_squared': r_squared
    }


def population_vector(rates, baselines, preferred_dirs, modulations):
    """Compute population vector from ensemble activity"""
    px, py = 0, 0

    for i in range(len(rates)):
        weight = (rates[i] - baselines[i]) / modulations[i]
        pd_rad = np.radians(preferred_dirs[i])
        px += weight * np.cos(pd_rad)
        py += weight * np.sin(pd_rad)

    decoded_direction = np.degrees(np.arctan2(py, px))
    magnitude = np.sqrt(px**2 + py**2)

    return decoded_direction, magnitude
