2. Bandpass Filtering

Filter raw signals to isolate spike band (300-6000 Hz).

15 min 
from scipy.signal import butter, filtfilt

def bandpass_filter(data, low, high, fs, order=4):
    """Apply Butterworth bandpass filter"""
    nyq = fs / 2
    b, a = butter(order, [low/nyq, high/nyq], btype='band')
    return filtfilt(b, a, data, axis=1)

# Filter for spikes (300-6000 Hz)
filtered = bandpass_filter(raw, 300, 6000, fs)

# Estimate noise level
noise_std = np.median(np.abs(filtered), axis=1) / 0.6745
print(f"Noise SD per channel: {noise_std[:5]}")

3. Spike Detection

Detect spikes using threshold crossing with dead time.

20 min 
def detect_spikes(signal, threshold, fs, dead_time_ms=1.0):
    """Detect spikes via threshold crossing"""
    dead_samples = int(dead_time_ms * fs / 1000)

    # Find threshold crossings
    crossings = np.where(signal < threshold)[0]

    # Apply dead time
    spike_times = []
    last_spike = -dead_samples
    for idx in crossings:
        if idx - last_spike >= dead_samples:
            spike_times.append(idx)
            last_spike = idx

    return np.array(spike_times)

# Detect on each channel
threshold = -4 * noise_std[0]
spikes_ch0 = detect_spikes(filtered[0], threshold, fs)
print(f"Detected {len(spikes_ch0)} spikes")

4. Feature Extraction with PCA

Extract spike waveforms and compute principal components.

25 min 
from sklearn.decomposition import PCA

def extract_waveforms(signal, spike_times, pre=10, post=30):
    """Extract spike waveforms around detection times"""
    waveforms = []
    for t in spike_times:
        if t - pre >= 0 and t + post < len(signal):
            waveforms.append(signal[t-pre:t+post])
    return np.array(waveforms)

# Extract waveforms
waveforms = extract_waveforms(filtered[0], spikes_ch0)

# PCA
pca = PCA(n_components=3)
features = pca.fit_transform(waveforms)

print(f"Variance explained: {pca.explained_variance_ratio_}")

        best_bic = gmm.bic(features)

6. Kalman Filter Decoder

Implement a Kalman filter for continuous velocity decoding.

45 min 
class KalmanDecoder:
    def __init__(self, state_dim, obs_dim):
        self.A = np.eye(state_dim)  # State transition
        self.H = np.zeros((obs_dim, state_dim))  # Observation
        self.W = np.eye(state_dim) * 0.01  # Process noise
        self.Q = np.eye(obs_dim) * 1.0  # Observation noise

    def fit(self, spike_rates, kinematics):
        """Fit decoder parameters via regression"""
        # H = (Y^T Y)^-1 Y^T X
        Y = spike_rates
        X = kinematics
        self.H = np.linalg.lstsq(Y, X, rcond=None)[0].T

        # Estimate noise covariances
        residuals = X - Y @ self.H.T
        self.Q = np.cov(residuals.T)

    def decode(self, y, x_prev, P_prev):
        """Single step Kalman update"""
        # Predict
        x_pred = self.A @ x_prev
        P_pred = self.A @ P_prev @ self.A.T + self.W

        # Update
        K = P_pred @ self.H.T @ np.linalg.inv(
            self.H @ P_pred @ self.H.T + self.Q)
        x = x_pred + K @ (y - self.H @ x_pred)
        P = (np.eye(len(x)) - K @ self.H) @ P_pred

        return x, P

from scipy.signal import lfilter, lfilter_zi

class RealtimeBCI:
    def __init__(self, decoder, fs=30000, bin_ms=50):
        self.decoder = decoder
        self.fs = fs
        self.bin_samples = int(bin_ms * fs / 1000)

        # Initialize causal filter
        b, a = butter(4, [300/15000, 6000/15000], btype='band')
        self.b, self.a = b, a
        self.zi = lfilter_zi(b, a)

        # Spike buffer
        self.spike_counts = np.zeros(n_units)
        self.sample_count = 0

    def process_sample(self, sample):
        """Process single sample, return velocity when bin complete"""
        # Causal filter
        filtered, self.zi = lfilter(self.b, self.a, [sample], zi=self.zi)

        # Online spike detection (simplified)
        if filtered[0] < self.threshold:
            self.spike_counts[0] += 1  # Would do sorting here

        self.sample_count += 1

        # Decode when bin completes
        if self.sample_count >= self.bin_samples:
            rates = self.spike_counts / (self.bin_samples / self.fs)
            velocity = self.decoder.decode(rates)
            self.spike_counts[:] = 0
            self.sample_count = 0
            return velocity
        return None
