import cv2
import numpy as np
import tensorflow as tf
from ultralytics import YOLO
import rospy
from sensor_msgs.msg import Image
from cv_bridge import CvBridge

class SelfDrivingVision:
    def __init__(self):
        # Initialize YOLO model
        self.model = YOLO('yolov8n.pt')
        self.bridge = CvBridge()
        
        # Initialize lane detection
        self.lane_detector = LaneDetector()
        
        # ROS subscribers and publishers
        rospy.init_node('self_driving_vision')
        self.image_sub = rospy.Subscriber('/camera/image_raw', Image, self.image_callback)
        self.detection_pub = rospy.Publisher('/vision/detections', DetectionArray, queue_size=1)
    
    def detect_objects(self, frame):
        """Detect objects in the frame using YOLO"""
        results = self.model(frame)
        detections = []
        
        for result in results:
            boxes = result.boxes
            for box in boxes:
                # Extract bounding box coordinates
                x1, y1, x2, y2 = box.xyxy[0].cpu().numpy()
                confidence = box.conf[0].cpu().numpy()
                class_id = int(box.cls[0].cpu().numpy())
                
                if confidence > 0.5:
                    detections.append({
                        'bbox': [x1, y1, x2, y2],
                        'confidence': confidence,
                        'class_id': class_id,
                        'class_name': self.model.names[class_id]
                    })
        
        return detections
    
    def detect_lanes(self, frame):
        """Detect lane boundaries using computer vision"""
        # Convert to grayscale
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        
        # Apply Gaussian blur
        blurred = cv2.GaussianBlur(gray, (5, 5), 0)
        
        # Edge detection
        edges = cv2.Canny(blurred, 50, 150)
        
        # Define region of interest
        height, width = edges.shape
        roi_vertices = np.array([[
            (0, height),
            (width//2, height//2),
            (width, height)
        ]], dtype=np.int32)
        
        roi = self.region_of_interest(edges, roi_vertices)
        
        # Hough line detection
        lines = cv2.HoughLinesP(roi, 1, np.pi/180, threshold=50,
                               minLineLength=50, maxLineGap=150)
        
        return self.extract_lane_lines(frame, lines)
    
    def predict_collision(self, detections, lanes):
        """Predict potential collision scenarios"""
        collision_risk = []
        
        for detection in detections:
            if detection['class_name'] in ['car', 'truck', 'bus', 'person']:
                # Calculate distance and relative velocity
                distance = self.calculate_distance(detection['bbox'])
                velocity = self.estimate_velocity(detection)
                
                # Collision prediction algorithm
                time_to_collision = distance / velocity if velocity > 0 else float('inf')
                
                if time_to_collision < 3.0:  # 3 seconds threshold
                    collision_risk.append({
                        'object': detection,
                        'time_to_collision': time_to_collision,
                        'risk_level': 'high' if time_to_collision < 1.0 else 'medium'
                    })
        
        return collision_risk
    
    def image_callback(self, msg):
        """ROS callback for processing incoming images"""
        try:
            # Convert ROS image to OpenCV format
            frame = self.bridge.imgmsg_to_cv2(msg, "bgr8")
            
            # Process the frame
            detections = self.detect_objects(frame)
            lanes = self.detect_lanes(frame)
            collision_risks = self.predict_collision(detections, lanes)
            
            # Publish results
            self.publish_detections(detections, lanes, collision_risks)
            
        except Exception as e:
            rospy.logerr(f"Error processing image: {str(e)}")
