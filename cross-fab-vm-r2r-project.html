Project Overview

Goal

Build a virtual metrology (VM) and run-to-run (R2R) advanced process control pipeline that predicts post-process CD and overlay per wafer/zone and automatically tunes recipes (feed-forward from litho → feedback to etch/depo). Show reduced CD 3σ and overlay error while cutting metrology sampling.

Key Achievements

  • VM MAE ≤ ±1.5 nm for post-etch CD (per zone)
  • Overlay MAE ≤ ±2 nm vs. sparse ground truth
  • CD 3σ reduction ≥ 25%; overlay P95 improvement ≥ 20%
  • 50–70% reduction in destructive/slow metrology sampling
  • Metrology queue time ↓ 30% (simulated fab)

Technical Architecture

Data Plumbing

Lithography Inputs (ASML-like):

  • Shot/field focus & dose
  • Slit/scan parameters
  • Stage temperatures
  • Alignment residuals
  • NILS proxy measurements

Etch/Deposition (Applied/AMAT-like):

  • Chamber IDs and states
  • RF power traces
  • Matching setpoints
  • Endpoint spectra
  • Pressure/flow/temperatures
  • Wafer edge ring state

Virtual Metrology

Models:

  • Quantile gradient boosting (CatBoost)
  • Conformal prediction for calibrated CIs

Features:

  • EWMA'd tool traces
  • Per-field focus/dose stats
  • Chamber drift counters
  • Wafer-position features (center/edge)

Targets:

  • Post-etch CD per zone
  • Overlay (nm) per field

R2R Control

Feed-forward:

  • Litho dose/focus → etch bias compensation

Feedback:

  • Double-EWMA / Kalman R2R
  • Adjust etch time/RF bias/depo rate
  • Optional litho dose trim next lot

Constraints:

  • Guard-bands
  • Chamber matching offsets
  • Recipe quantization

SPC + FDC Integration

  • Multivariate T² / SPE charts
  • Drift detection → automatic chamber de-rate
  • Recipe hold capability
  • Root-cause hints via SHAP

import numpy as np
import pandas as pd
from dataclasses import dataclass
from typing import Dict, List, Optional
import json

@dataclass
class LithoData:
    """ASML-like lithography data structure"""
    wafer_id: str
    shot_map: np.ndarray  # Focus/dose per shot
    field_data: Dict[str, float]  # Slit/scan parameters
    stage_temps: np.ndarray
    alignment_residuals: Dict[str, float]
    nils_proxy: float
    timestamp: float

@dataclass
class EtchData:
    """Applied/AMAT-like etch data structure"""
    wafer_id: str
    chamber_id: str
    rf_power_trace: np.ndarray
    matching_setpoints: Dict[str, float]
    endpoint_spectra: np.ndarray
    process_params: Dict[str, float]  # Pressure, flows, temps
    edge_ring_state: str
    timestamp: float

class FabDataSimulator:
    """Simulates realistic fab data with correlations"""
    
    def __init__(self, config_file: str = "fab_config.json"):
        with open(config_file, 'r') as f:
            self.config = json.load(f)
        
        self.chamber_drift = {}
        self.reticle_aging = {}
        
    def generate_litho_batch(self, batch_size: int = 25) -> List[LithoData]:
        """Generate a batch of lithography data"""
        batch = []
        
        for i in range(batch_size):
            # Simulate focus/dose variation with systematic + random components
            focus_systematic = np.random.normal(0, 0.05)  # nm
            dose_systematic = np.random.normal(0, 0.5)    # mJ/cm²
            
            shot_map = np.random.normal(
                [focus_systematic, dose_systematic + 20], 
                [0.02, 0.2], 
                size=(33, 2)  # 33 shots per field
            )
            
            # Stage temperature drift
            stage_temps = np.random.normal(23.0, 0.1, size=4)  # °C
            
            # Alignment residuals (nm)
            alignment = {
                'x_residual': np.random.normal(0, 2.0),
                'y_residual': np.random.normal(0, 2.0),
                'rotation': np.random.normal(0, 0.001)  # mrad
            }
            
            batch.append(LithoData(
                wafer_id=f"W{i:03d}",
                shot_map=shot_map,
                field_data={'slit_width': 26.0, 'scan_speed': 500},
                stage_temps=stage_temps,
                alignment_residuals=alignment,
                nils_proxy=np.random.normal(2.5, 0.1),
                timestamp=i * 300  # 5 min per wafer
            ))
            
        return batch
    
    def generate_etch_batch(self, litho_batch: List[LithoData]) -> List[EtchData]:
        """Generate corresponding etch data with process correlations"""
        etch_batch = []
        
        for litho in litho_batch:
            chamber_id = np.random.choice(['CH01', 'CH02', 'CH03'])
            
            # Chamber drift simulation
            if chamber_id not in self.chamber_drift:
                self.chamber_drift[chamber_id] = 0.0
            
            self.chamber_drift[chamber_id] += np.random.normal(0, 0.1)
            
            # RF power trace with endpoint detection
            base_power = 2000 + self.chamber_drift[chamber_id] * 10
            power_trace = base_power + np.random.normal(0, 50, size=180)  # 3 min trace
            
            # Endpoint spectra simulation
            endpoint_spectra = self._simulate_endpoint_spectra()
            
            etch_batch.append(EtchData(
                wafer_id=litho.wafer_id,
                chamber_id=chamber_id,
                rf_power_trace=power_trace,
                matching_setpoints={'C1': 500, 'C2': 300},
                endpoint_spectra=endpoint_spectra,
                process_params={
                    'pressure': np.random.normal(15.0, 0.5),  # mTorr
                    'cf4_flow': np.random.normal(100, 2),     # sccm
                    'temperature': np.random.normal(60, 2)     # °C
                },
                edge_ring_state='NORMAL',
                timestamp=litho.timestamp + 1800  # 30 min later
            ))
            
        return etch_batch
    
    def _simulate_endpoint_spectra(self) -> np.ndarray:
        """Simulate OES endpoint detection spectra"""
        wavelengths = np.linspace(200, 800, 600)
        
        # CF4 lines at specific wavelengths
        cf4_lines = [250, 280, 320, 450]
        spectrum = np.zeros_like(wavelengths)
        
        for line in cf4_lines:
            intensity = np.random.normal(1000, 100)
            width = np.random.normal(2, 0.2)
            spectrum += intensity * np.exp(-(wavelengths - line)**2 / (2 * width**2))
        
        # Add noise
        spectrum += np.random.normal(0, 50, size=len(wavelengths))
        
        return spectrum

Virtual Metrology Model


import catboost as cb
from sklearn.base import BaseEstimator, RegressorMixin
from sklearn.model_selection import cross_val_score
import numpy as np
import pandas as pd
from typing import Tuple, Dict
import joblib

class VirtualMetrologyModel(BaseEstimator, RegressorMixin):
    """
    Virtual Metrology model using CatBoost with conformal prediction
    for calibrated confidence intervals
    """
    
    def __init__(self, 
                 iterations: int = 1000,
                 learning_rate: float = 0.1,
                 depth: int = 6,
                 alpha: float = 0.1):  # For 90% CI
        
        self.iterations = iterations
        self.learning_rate = learning_rate  
        self.depth = depth
        self.alpha = alpha
        
        self.model = None
        self.conformal_lower = None
        self.conformal_upper = None
        self.feature_names = None
        self.is_fitted = False
        
    def _extract_features(self, litho_data: List[LithoData], 
                         etch_data: List[EtchData]) -> pd.DataFrame:
        """Extract engineered features from raw fab data"""
        
        features = []
        
        for litho, etch in zip(litho_data, etch_data):
            # Litho features
            focus_stats = {
                'focus_mean': np.mean(litho.shot_map[:, 0]),
                'focus_std': np.std(litho.shot_map[:, 0]),
                'focus_range': np.ptp(litho.shot_map[:, 0]),
                'focus_center': litho.shot_map[16, 0],  # Center shot
                'focus_edge_avg': np.mean([litho.shot_map[0, 0], litho.shot_map[-1, 0]])
            }
            
            dose_stats = {
                'dose_mean': np.mean(litho.shot_map[:, 1]),
                'dose_std': np.std(litho.shot_map[:, 1]),
                'dose_uniformity': np.std(litho.shot_map[:, 1]) / np.mean(litho.shot_map[:, 1])
            }
            
            # Stage and alignment features
            stage_features = {
                'stage_temp_avg': np.mean(litho.stage_temps),
                'stage_temp_delta': np.max(litho.stage_temps) - np.min(litho.stage_temps),
                'alignment_x': litho.alignment_residuals['x_residual'],
                'alignment_y': litho.alignment_residuals['y_residual'],
                'alignment_rotation': litho.alignment_residuals['rotation'],
                'nils_proxy': litho.nils_proxy
            }
            
            # Etch features with EWMA
            rf_features = {
                'rf_power_mean': np.mean(etch.rf_power_trace),
                'rf_power_std': np.std(etch.rf_power_trace),
                'rf_power_trend': np.polyfit(range(len(etch.rf_power_trace)), 
                                           etch.rf_power_trace, 1)[0],
                'chamber_id': etch.chamber_id
            }
            
            # Process parameter features
            process_features = {
                f'process_{k}': v for k, v in etch.process_params.items()
            }
            
            # Endpoint detection features
            endpoint_features = {
                'endpoint_peak': np.max(etch.endpoint_spectra),
                'endpoint_integral': np.trapz(etch.endpoint_spectra),
                'endpoint_cf4_ratio': np.sum(etch.endpoint_spectra[240:260]) / 
                                    np.sum(etch.endpoint_spectra)
            }
            
            # Combine all features
            wafer_features = {
                'wafer_id': litho.wafer_id,
                **focus_stats,
                **dose_stats, 
                **stage_features,
                **rf_features,
                **process_features,
                **endpoint_features
            }
            
            features.append(wafer_features)
            
        return pd.DataFrame(features)
    
    def fit(self, litho_data: List[LithoData], etch_data: List[EtchData], 
            cd_targets: np.ndarray, overlay_targets: np.ndarray):
        """
        Fit the virtual metrology model
        
        Args:
            litho_data: Lithography process data
            etch_data: Etch process data  
            cd_targets: Post-etch CD measurements (nm)
            overlay_targets: Overlay measurements (nm)
        """
        
        # Extract features
        X = self._extract_features(litho_data, etch_data)
        
        # Separate categorical and numerical features
        categorical_features = ['chamber_id']
        numerical_features = [col for col in X.columns 
                            if col not in categorical_features + ['wafer_id']]
        
        X_model = X[categorical_features + numerical_features].copy()
        self.feature_names = X_model.columns.tolist()
        
        # Fit CD model
        self.cd_model = cb.CatBoostRegressor(
            iterations=self.iterations,
            learning_rate=self.learning_rate,
            depth=self.depth,
            cat_features=categorical_features,
            verbose=False,
            random_seed=42
        )
        
        self.cd_model.fit(X_model, cd_targets)
        
        # Fit overlay model  
        self.overlay_model = cb.CatBoostRegressor(
            iterations=self.iterations,
            learning_rate=self.learning_rate,
            depth=self.depth,
            cat_features=categorical_features,
            verbose=False,
            random_seed=42
        )
        
        self.overlay_model.fit(X_model, overlay_targets)
        
        # Conformal prediction calibration
        self._calibrate_conformal_prediction(X_model, cd_targets, overlay_targets)
        
        self.is_fitted = True
        
        return self
    
    def _calibrate_conformal_prediction(self, X: pd.DataFrame, 
                                      cd_targets: np.ndarray,
                                      overlay_targets: np.ndarray):
        """Calibrate conformal prediction intervals"""
        
        # Split calibration set (20% of data)
        n_cal = int(0.2 * len(X))
        cal_idx = np.random.choice(len(X), n_cal, replace=False)
        
        X_cal = X.iloc[cal_idx]
        cd_cal = cd_targets[cal_idx]
        overlay_cal = overlay_targets[cal_idx]
        
        # Get predictions on calibration set
        cd_pred_cal = self.cd_model.predict(X_cal)
        overlay_pred_cal = self.overlay_model.predict(X_cal)
        
        # Compute nonconformity scores
        cd_scores = np.abs(cd_cal - cd_pred_cal)
        overlay_scores = np.abs(overlay_cal - overlay_pred_cal)
        
        # Compute quantiles for confidence intervals
        self.cd_quantile = np.quantile(cd_scores, 1 - self.alpha)
        self.overlay_quantile = np.quantile(overlay_scores, 1 - self.alpha)
    
    def predict(self, litho_data: List[LithoData], 
                etch_data: List[EtchData]) -> Dict[str, np.ndarray]:
        """
        Predict CD and overlay with confidence intervals
        
        Returns:
            Dictionary with predictions and confidence intervals
        """
        
        if not self.is_fitted:
            raise ValueError("Model must be fitted before prediction")
            
        # Extract features
        X = self._extract_features(litho_data, etch_data)
        X_model = X[self.feature_names]
        
        # Get point predictions
        cd_pred = self.cd_model.predict(X_model)
        overlay_pred = self.overlay_model.predict(X_model)
        
        # Compute confidence intervals
        cd_lower = cd_pred - self.cd_quantile
        cd_upper = cd_pred + self.cd_quantile
        
        overlay_lower = overlay_pred - self.overlay_quantile  
        overlay_upper = overlay_pred + self.overlay_quantile
        
        return {
            'cd_prediction': cd_pred,
            'cd_lower': cd_lower,
            'cd_upper': cd_upper,
            'overlay_prediction': overlay_pred,
            'overlay_lower': overlay_lower,
            'overlay_upper': overlay_upper,
            'wafer_ids': X['wafer_id'].values
        }
    
    def get_feature_importance(self) -> Dict[str, float]:
        """Get feature importance from trained models"""
        
        if not self.is_fitted:
            raise ValueError("Model must be fitted first")
            
        cd_importance = dict(zip(self.feature_names, 
                               self.cd_model.feature_importances_))
        overlay_importance = dict(zip(self.feature_names,
                                    self.overlay_model.feature_importances_))
        
        return {
            'cd_importance': cd_importance,
            'overlay_importance': overlay_importance
        }
    
    def save_model(self, filepath: str):
        """Save trained model"""
        model_data = {
            'cd_model': self.cd_model,
            'overlay_model': self.overlay_model,
            'cd_quantile': self.cd_quantile,
            'overlay_quantile': self.overlay_quantile,
            'feature_names': self.feature_names,
            'alpha': self.alpha
        }
        
        joblib.dump(model_data, filepath)
    
    def load_model(self, filepath: str):
        """Load trained model"""
        model_data = joblib.load(filepath)
        
        self.cd_model = model_data['cd_model']
        self.overlay_model = model_data['overlay_model'] 
        self.cd_quantile = model_data['cd_quantile']
        self.overlay_quantile = model_data['overlay_quantile']
        self.feature_names = model_data['feature_names']
        self.alpha = model_data['alpha']
        self.is_fitted = True

Run-to-Run APC Controller


import numpy as np
from typing import Dict, List, Optional, Tuple
from dataclasses import dataclass, field
from abc import ABC, abstractmethod
import logging

@dataclass
class ControlAction:
    """Represents a control action to be applied"""
    recipe_id: str
    parameter_adjustments: Dict[str, float]
    timestamp: float
    confidence: float
    rationale: str

@dataclass 
class ProcessState:
    """Current state of the process"""
    cd_measurement: Optional[float] = None
    overlay_measurement: Optional[float] = None
    cd_prediction: Optional[float] = None
    overlay_prediction: Optional[float] = None
    cd_target: float = 45.0  # nm
    overlay_target: float = 0.0  # nm
    wafer_id: str = ""
    timestamp: float = 0.0

class R2RController(ABC):
    """Abstract base class for Run-to-Run controllers"""
    
    @abstractmethod
    def update(self, state: ProcessState) -> Optional[ControlAction]:
        pass
    
    @abstractmethod
    def reset(self):
        pass

class DoubleEWMAController(R2RController):
    """
    Double EWMA Run-to-Run Controller
    Implements the standard semiconductor R2R control algorithm
    """
    
    def __init__(self, 
                 alpha: float = 0.2,      # Discount factor for level
                 beta: float = 0.1,       # Discount factor for trend  
                 cd_gain: float = 0.8,    # CD control gain
                 overlay_gain: float = 0.6, # Overlay control gain
                 constraints: Optional[Dict] = None):
        
        self.alpha = alpha
        self.beta = beta
        self.cd_gain = cd_gain
        self.overlay_gain = overlay_gain
        self.constraints = constraints or self._default_constraints()
        
        # State variables
        self.cd_level = 0.0      # Current level estimate
        self.cd_trend = 0.0      # Current trend estimate  
        self.overlay_level = 0.0
        self.overlay_trend = 0.0
        
        self.history = []
        self.initialized = False
        
        self.logger = logging.getLogger(__name__)
    
    def _default_constraints(self) -> Dict:
        """Default process constraints"""
        return {
            'etch_time': {'min': 60, 'max': 180},      # seconds
            'rf_bias': {'min': -50, 'max': 50},        # watts  
            'dose_adjustment': {'min': -2, 'max': 2},   # mJ/cm²
            'focus_adjustment': {'min': -0.1, 'max': 0.1} # µm
        }
    
    def update(self, state: ProcessState) -> Optional[ControlAction]:
        """
        Update controller with new measurement and return control action
        """
        
        # Use measurement if available, otherwise use prediction
        cd_error = (state.cd_measurement or state.cd_prediction) - state.cd_target
        overlay_error = (state.overlay_measurement or state.overlay_prediction) - state.overlay_target
        
        if not self.initialized:
            # Initialize with first measurement
            self.cd_level = cd_error
            self.overlay_level = overlay_error
            self.initialized = True
            self.logger.info(f"Controller initialized with CD error: {cd_error:.2f}, Overlay error: {overlay_error:.2f}")
            return None
        
        # Update CD estimates
        prev_cd_level = self.cd_level
        self.cd_level = self.alpha * cd_error + (1 - self.alpha) * (self.cd_level + self.cd_trend)
        self.cd_trend = self.beta * (self.cd_level - prev_cd_level) + (1 - self.beta) * self.cd_trend
        
        # Update overlay estimates  
        prev_overlay_level = self.overlay_level
        self.overlay_level = self.alpha * overlay_error + (1 - self.alpha) * (self.overlay_level + self.overlay_trend)
        self.overlay_trend = self.beta * (self.overlay_level - prev_overlay_level) + (1 - self.beta) * self.overlay_trend
        
        # Calculate control actions
        cd_forecast = self.cd_level + self.cd_trend
        overlay_forecast = self.overlay_level + self.overlay_trend
        
        # Control actions (negative feedback)
        etch_time_adj = -self.cd_gain * cd_forecast * 2.0  # 2 sec/nm gain
        rf_bias_adj = -self.cd_gain * cd_forecast * 0.5    # 0.5 W/nm gain  
        
        dose_adj = -self.overlay_gain * overlay_forecast * 0.1  # 0.1 mJ/cm²/nm
        focus_adj = -self.overlay_gain * overlay_forecast * 0.01 # 0.01 µm/nm
        
        # Apply constraints
        adjustments = {
            'etch_time': self._apply_constraints(etch_time_adj, 'etch_time'),
            'rf_bias': self._apply_constraints(rf_bias_adj, 'rf_bias'),
            'dose_adjustment': self._apply_constraints(dose_adj, 'dose_adjustment'),  
            'focus_adjustment': self._apply_constraints(focus_adj, 'focus_adjustment')
        }
        
        # Calculate confidence based on prediction uncertainty
        confidence = self._calculate_confidence(state)
        
        # Create control action
        action = ControlAction(
            recipe_id=f"R2R_{state.wafer_id}",
            parameter_adjustments=adjustments,
            timestamp=state.timestamp,
            confidence=confidence,
            rationale=f"CD forecast: {cd_forecast:.2f}nm, Overlay forecast: {overlay_forecast:.2f}nm"
        )
        
        # Log control action
        self.logger.info(f"Control action for {state.wafer_id}: {adjustments}")
        self.history.append((state, action))
        
        return action
    
    def _apply_constraints(self, value: float, param_name: str) -> float:
        """Apply process constraints to control action"""
        constraints = self.constraints.get(param_name, {})
        min_val = constraints.get('min', float('-inf'))
        max_val = constraints.get('max', float('inf'))
        
        return np.clip(value, min_val, max_val)
    
    def _calculate_confidence(self, state: ProcessState) -> float:
        """Calculate confidence in control action based on state"""
        
        # Higher confidence when we have actual measurements
        base_confidence = 0.9 if state.cd_measurement is not None else 0.7
        
        # Reduce confidence if we're near constraints
        constraint_penalty = 0.0
        # (Implementation would check how close we are to process limits)
        
        return max(0.1, base_confidence - constraint_penalty)
    
    def reset(self):
        """Reset controller state"""
        self.cd_level = 0.0
        self.cd_trend = 0.0
        self.overlay_level = 0.0
        self.overlay_trend = 0.0
        self.history = []
        self.initialized = False
        self.logger.info("Controller reset")
    
    def get_state(self) -> Dict:
        """Get current controller state for monitoring"""
        return {
            'cd_level': self.cd_level,
            'cd_trend': self.cd_trend,
            'overlay_level': self.overlay_level,
            'overlay_trend': self.overlay_trend,
            'initialized': self.initialized,
            'history_length': len(self.history)
        }

class KalmanR2RController(R2RController):
    """
    Kalman Filter based R2R Controller
    More sophisticated state estimation with process noise modeling
    """
    
    def __init__(self,
                 process_noise_cd: float = 0.5,
                 process_noise_overlay: float = 0.3,
                 measurement_noise_cd: float = 1.0,
                 measurement_noise_overlay: float = 1.5):
        
        # State: [cd_bias, cd_trend, overlay_bias, overlay_trend]
        self.state = np.zeros(4)
        
        # Process noise covariance
        Q = np.diag([process_noise_cd**2, 0.1, process_noise_overlay**2, 0.1])
        
        # Measurement noise covariance  
        R = np.diag([measurement_noise_cd**2, measurement_noise_overlay**2])
        
        # State transition matrix (constant velocity model)
        self.F = np.array([
            [1, 1, 0, 0],  # cd_bias(k+1) = cd_bias(k) + cd_trend(k)
            [0, 1, 0, 0],  # cd_trend(k+1) = cd_trend(k)
            [0, 0, 1, 1],  # overlay_bias(k+1) = overlay_bias(k) + overlay_trend(k)
            [0, 0, 0, 1]   # overlay_trend(k+1) = overlay_trend(k)
        ])
        
        # Measurement matrix
        self.H = np.array([
            [1, 0, 0, 0],  # We observe cd_bias
            [0, 0, 1, 0]   # We observe overlay_bias
        ])
        
        # Error covariance matrix
        self.P = np.eye(4) * 10.0  # Initial uncertainty
        
        self.Q = Q
        self.R = R
        
        self.history = []
        self.logger = logging.getLogger(__name__)
    
    def update(self, state: ProcessState) -> Optional[ControlAction]:
        """Kalman filter update step"""
        
        # Prediction step
        self.state = self.F @ self.state
        self.P = self.F @ self.P @ self.F.T + self.Q
        
        # Measurement update (if we have measurements)
        cd_obs = state.cd_measurement or state.cd_prediction
        overlay_obs = state.overlay_measurement or state.overlay_prediction
        
        if cd_obs is not None and overlay_obs is not None:
            # Measurement residual
            z = np.array([cd_obs - state.cd_target, overlay_obs - state.overlay_target])
            y = z - self.H @ self.state
            
            # Kalman gain
            S = self.H @ self.P @ self.H.T + self.R
            K = self.P @ self.H.T @ np.linalg.inv(S)
            
            # Update state and covariance
            self.state = self.state + K @ y
            self.P = (np.eye(4) - K @ self.H) @ self.P
        
        # Generate control action based on predicted state
        cd_forecast = self.state[0] + self.state[1]  # bias + trend
        overlay_forecast = self.state[2] + self.state[3]
        
        # Control gains (negative feedback)
        adjustments = {
            'etch_time': -0.8 * cd_forecast * 2.0,
            'rf_bias': -0.8 * cd_forecast * 0.5,
            'dose_adjustment': -0.6 * overlay_forecast * 0.1,
            'focus_adjustment': -0.6 * overlay_forecast * 0.01
        }
        
        # Calculate confidence from state uncertainty
        cd_uncertainty = np.sqrt(self.P[0, 0])
        overlay_uncertainty = np.sqrt(self.P[2, 2])
        confidence = 1.0 / (1.0 + cd_uncertainty + overlay_uncertainty)
        
        action = ControlAction(
            recipe_id=f"Kalman_{state.wafer_id}",
            parameter_adjustments=adjustments,
            timestamp=state.timestamp,
            confidence=confidence,
            rationale=f"Kalman forecast - CD: {cd_forecast:.2f}nm, Overlay: {overlay_forecast:.2f}nm"
        )
        
        self.history.append((state, action))
        return action
    
    def reset(self):
        """Reset Kalman filter"""
        self.state = np.zeros(4)
        self.P = np.eye(4) * 10.0
        self.history = []

    def generate_wafer_map_data(self, metric_type):
        """Generate simulated wafer map data"""
        # 33 sites in a typical wafer map pattern
        sites = []
        for i in range(-3, 4):
            for j in range(-3, 4):
                if i*i + j*j <= 9:  # Circular wafer
                    if metric_type == "CD":
                        value = 45.0 + np.random.normal(0, 1.5) + 0.5 * (i*i + j*j)**0.5
                    else:  # Overlay
                        value = np.random.normal(0, 2.0) + 0.3 * i + 0.2 * j
                    
                    sites.append({
                        'x': j, 'y': i, 'value': value,
                        'site_id': f"S{len(sites)+1:02d}"
                    })
        
        return pd.DataFrame(sites)
    
    def create_wafer_map(self, data, title, colorscale):
        """Create wafer map visualization"""
        fig = go.Figure(data=go.Scatter(
            x=data['x'],
            y=data['y'], 
            mode='markers',
            marker=dict(
                size=30,
                color=data['value'],
                colorscale=colorscale,
                showscale=True,
                colorbar=dict(title=title)
            ),
            text=data['site_id'],
            hovertemplate=f'Site: %{{text}}
{title}: %{{marker.color:.2f}}'
        ))
        
        # Add wafer outline
        theta = np.linspace(0, 2*np.pi, 100)
        x_circle = 3.5 * np.cos(theta)
        y_circle = 3.5 * np.sin(theta)
        
        fig.add_trace(go.Scatter(
            x=x_circle, y=y_circle,
            mode='lines',
            line=dict(color='black', width=2),
            showlegend=False,
            hoverinfo='skip'
        ))
        
        fig.update_layout(
            title=title,
            xaxis=dict(scaleanchor="y", scaleratio=1, showgrid=False, zeroline=False),
            yaxis=dict(showgrid=False, zeroline=False),
            height=400
        )
        
        return fig
